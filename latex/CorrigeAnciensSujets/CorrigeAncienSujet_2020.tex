\textbf{Exercice 1} \quad $A=\begin{pmatrix}1&0&-1\\0&1&0\\-1&1&1\end{pmatrix}$.

\textbf{1. Inversibilité.} En développant suivant la deuxième ligne $(0,1,0)$ :
\[
\det A=1\times\begin{vmatrix}1&-1\\-1&1\end{vmatrix}=1-1=0 .
\]
Donc \textbf{$A$ n'est pas inversible}.
\remarque{L'énoncé original demandait de « justifier que $A$ est inversible » ; c'est faux, comme le confirme la valeur propre $0$ trouvée à la question 3.}

\textbf{2. Polynôme caractéristique.} En développant suivant la deuxième ligne $(0,1-x,0)$ :
\[
P_A(x)=\det(A-xI_3)=(1-x)\begin{vmatrix}1-x&-1\\-1&1-x\end{vmatrix}=(1-x)\big[(1-x)^2-1\big]=(1-x)(-x)(2-x),
\]
\[
\boxed{P_A(x)=-x(1-x)(2-x)=-x(x-1)(x-2)}.
\]

\textbf{3. Valeurs propres} (racines de $P_A$, dans l'ordre croissant) : $\boxed{\alpha=0,\ \beta=1,\ \gamma=2}$.

\textbf{4. Vecteurs propres.}
\begin{itemize}
\item $\alpha=0$ : $AX=0\iff\begin{cases}x-z=0\\y=0\\-x+y+z=0\end{cases}\iff y=0,\ z=x$. On prend $u=(1,0,1)$.
\item $\beta=1$ : $(A-I)X=0\iff\begin{cases}-z=0\\-x+y=0\end{cases}$. On prend $v=(1,1,0)$.
\item $\gamma=2$ : $(A-2I)X=0\iff\begin{cases}-x-z=0\\-y=0\\-x+y-z=0\end{cases}\iff y=0,\ z=-x$. On prend $w=(1,0,-1)$.
\end{itemize}
(Vérification : $Au=(0,0,0)$, $Av=(1,1,0)=v$, $Aw=(2,0,-2)=2w$.)

\textbf{5. Matrice de passage.} Les trois valeurs propres étant distinctes, $A$ est diagonalisable et $(u,v,w)$ est une base de vecteurs propres. La matrice de passage de la base canonique à la base $(u,v,w)$ a pour colonnes $u,v,w$ :
\[
\boxed{P=\begin{pmatrix}1&1&1\\0&1&0\\1&0&-1\end{pmatrix}},\qquad A=PDP^{-1},\qquad D=\begin{pmatrix}0&0&0\\0&1&0\\0&0&2\end{pmatrix}.
\]
($\det P=-2\neq0$ ; on trouve $P^{-1}=\frac12\begin{pmatrix}1&-1&1\\0&2&0\\1&-1&-1\end{pmatrix}$.)

\medskip\textbf{Exercice 2}

\textbf{1.a)} $f(x)=e^{-x}$ est la composée de $x\mapsto-x$ (dérivable sur $\mathbb{R}$) et de $\exp$ (dérivable sur $\mathbb{R}$) : $f$ est dérivable sur $\mathbb{R}$ (et même de classe $C^\infty$).

\textbf{1.b)} $f'(x)=-e^{-x}$, \quad $f''(x)=e^{-x}$, \quad $f^{(3)}(x)=-e^{-x}$.

\textbf{1.c) Récurrence.} Soit $\mathcal{P}(n)$ : « $f^{(n)}(x)=(-1)^ne^{-x}$ pour tout $x$ ».
\emph{Initialisation} : $f^{(0)}=f=(-1)^0e^{-x}$.
\emph{Hérédité} : si $f^{(n)}(x)=(-1)^ne^{-x}$, alors $f^{(n+1)}(x)=\big((-1)^ne^{-x}\big)'=(-1)^n\cdot(-e^{-x})=(-1)^{n+1}e^{-x}$.
Donc $\mathcal{P}(n)$ est vraie pour tout $n\in\mathbb{N}$.

\textbf{1.d) Développement limité.} $f$ est de classe $C^n$ ; la formule de Taylor-Young en $0$ donne, avec $f^{(k)}(0)=(-1)^k$ :
\[
e^{-x}=\sum_{k=0}^n\frac{f^{(k)}(0)}{k!}x^k+o(x^n)=\boxed{1-x+\frac{x^2}{2!}-\frac{x^3}{3!}+\dots+(-1)^n\frac{x^n}{n!}+o(x^n)}.
\]

\textbf{2.} $(\Gamma)$ : $x(t)=\cos3t$, $y(t)=\sin2t$. \emph{(C'est la même courbe que dans l'exercice 1 de la session 2016.)}

\textbf{2.a) Réduction de l'intervalle d'étude.}

\underline{Périodicité.} $x$ est $\frac{2\pi}{3}$-périodique et $y$ est $\pi$-périodique ; elles sont donc toutes deux $2\pi$-périodiques ($2\pi=3\times\frac{2\pi}3=2\times\pi$). Le point $M(t+2\pi)$ est égal à $M(t)$ : il suffit d'étudier $t\in[-\pi,\pi]$.

\underline{Parité.} $x(-t)=\cos(-3t)=\cos3t=x(t)$ et $y(-t)=\sin(-2t)=-\sin2t=-y(t)$. Le point $M(-t)$ est le symétrique de $M(t)$ par rapport à l'axe des abscisses $(Ox)$.

Il suffit donc d'étudier la courbe pour $t\in[0,\pi]$, puis de compléter par la symétrie d'axe $(Ox)$ : on obtient ainsi la courbe entière.

\textbf{2.b) Tableau de variation commun de $x$ et $y$ sur $[0,\pi]$.}

\underline{$x'(t)=-3\sin3t$.} Sur $[0,\pi]$, $3t\in[0,3\pi]$ : $\sin3t\ge0$ pour $t\in[0,\frac\pi3]\cup[\frac{2\pi}3,\pi]$ et $\sin3t\le0$ pour $t\in[\frac\pi3,\frac{2\pi}3]$. Donc $x'\le0$ sur $[0,\frac\pi3]$ et sur $[\frac{2\pi}3,\pi]$, $x'\ge0$ sur $[\frac\pi3,\frac{2\pi}3]$ ; $x'$ s'annule en $0,\frac\pi3,\frac{2\pi}3,\pi$.

\underline{$y'(t)=2\cos2t$.} Sur $[0,\pi]$, $2t\in[0,2\pi]$ : $y'\ge0$ sur $[0,\frac\pi4]$ et sur $[\frac{3\pi}4,\pi]$, $y'\le0$ sur $[\frac\pi4,\frac{3\pi}4]$ ; $y'$ s'annule en $\frac\pi4$ et $\frac{3\pi}4$.

\medskip\textbf{Exercice 3} \quad $\beta(P,Q)=\displaystyle\int_0^1xP(x)Q'(x)\,dx$ sur $\mathbb{R}_n[X]$.

\textbf{1. $\beta$ est bilinéaire.} Soient $P,G,Q\in\mathbb{R}_n[X]$ et $\lambda\in\mathbb{R}$.
\begin{itemize}
\item \emph{Linéarité à gauche} : par linéarité de l'intégrale,
\[
\beta(P+\lambda G,Q)=\int_0^1x\big(P(x)+\lambda G(x)\big)Q'(x)dx=\beta(P,Q)+\lambda\beta(G,Q).
\]
\item \emph{Linéarité à droite} : la dérivation est linéaire, $(Q+\lambda G)'=Q'+\lambda G'$, donc
\[
\beta(P,Q+\lambda G)=\int_0^1xP(x)\big(Q'(x)+\lambda G'(x)\big)dx=\beta(P,Q)+\lambda\beta(P,G).
\]
\end{itemize}
$\beta$ est une forme (à valeurs réelles) bilinéaire sur $\mathbb{R}_n[X]$.

\textbf{2.} Avec $P=X$, $Q=1$ ($Q'=0$) puis $P=1$, $Q=X$ ($Q'=1$) :
\[
\beta(X,1)=\int_0^1x\cdot x\cdot0\,dx=\boxed{0},\qquad\beta(1,X)=\int_0^1x\cdot1\cdot1\,dx=\Big[\frac{x^2}{2}\Big]_0^1=\boxed{\frac12}.
\]

\textbf{3. Symétrie, antisymétrie.}
\begin{itemize}
\item $\beta(X,1)=0\neq\frac12=\beta(1,X)$ : $\beta$ \textbf{n'est pas symétrique}.
\item $\beta$ antisymétrique signifierait $\beta(P,Q)=-\beta(Q,P)$ pour tous $P,Q$. Or $\beta(X,1)=0\neq-\frac12=-\beta(1,X)$ (ou encore $\beta(X,X)=\int_0^1x^2dx=\frac13\neq0$). Donc $\beta$ \textbf{n'est pas antisymétrique} non plus.
\end{itemize}

\medskip\textbf{Exercice 4}

\textbf{1. Intégrale de Gauss.} $x\mapsto e^{-x^2/2}$ est continue, positive, et $e^{-x^2/2}\le e^{-x+1/2}$ pour $x\ge0$ (car $\frac{x^2}{2}-x+\frac12=\frac{(x-1)^2}{2}\ge0$), donc $I$ converge. Par le théorème de Fubini (fonctions positives) :
\[
I^2=\int_0^{+\infty}e^{-x^2/2}dx\int_0^{+\infty}e^{-y^2/2}dy=\iint_{[0,+\infty[^2}e^{-\frac{x^2+y^2}{2}}\,dx\,dy .
\]
En coordonnées polaires $x=r\cos\theta$, $y=r\sin\theta$ (jacobien $r$), le quart de plan $x\ge0,y\ge0$ correspond à $r\ge0$, $\theta\in[0,\frac\pi2]$ :
\[
I^2=\int_0^{\pi/2}d\theta\int_0^{+\infty}re^{-r^2/2}dr=\frac\pi2\Big[-e^{-r^2/2}\Big]_0^{+\infty}=\frac\pi2 .
\]
Comme $I>0$ : $\boxed{I=\sqrt{\dfrac\pi2}}$.

\textbf{2.} $f_n(x)=\Big(1+\dfrac{x^2}{n}\Big)^{-\frac{n+1}{2}}$ ($n\ge1$) et $a_n=\displaystyle\int_{-\infty}^{+\infty}f_n(x)\,dx$.

\remarque{L'énoncé original écrivait $f_n(x)=(1+\frac{x^2}{n})-\frac{n+1}{2}$ : il s'agit évidemment d'un exposant $-\frac{n+1}{2}$ (sinon $a_n$ n'existe pas).}

\textbf{a) Inégalité de Bernoulli : $(1+u)^p\ge1+pu$ pour $u\ge-1$ et $p\in\mathbb{N}$.} Récurrence sur $p$. Pour $p=0$ : $1\ge1$. Si $(1+u)^p\ge1+pu$, comme $1+u\ge0$ on peut multiplier par $1+u$ :
\[
(1+u)^{p+1}\ge(1+pu)(1+u)=1+(p+1)u+pu^2\ge1+(p+1)u .
\]
(L'hypothèse $u\ge-1$ est nécessaire : pour $p=3$ et $u=-4$, $(1+u)^3=-27<1+3u=-11$.)

\textbf{b) $f_n(x)\le\dfrac{2}{1+x^2}$.} Soit $u=\frac{x^2}{n}\ge0$ et $q=\big\lfloor\frac{n+1}{2}\big\rfloor$, entier vérifiant $q\ge\frac n2$ et $q\le\frac{n+1}{2}$. Comme $1+u\ge1$ :
\[
\Big(1+\frac{x^2}{n}\Big)^{\frac{n+1}{2}}\ge(1+u)^q\ge1+qu\ge1+\frac n2\cdot\frac{x^2}{n}=1+\frac{x^2}{2}.
\]
En passant à l'inverse : $f_n(x)\le\dfrac{1}{1+\frac{x^2}{2}}=\dfrac{2}{2+x^2}\le\dfrac{2}{1+x^2}$.

\textbf{c) Limite de $a_n$.}
\begin{itemize}
\item \emph{Convergence simple} : pour $x$ fixé, $\ln f_n(x)=-\frac{n+1}{2}\ln\big(1+\frac{x^2}{n}\big)$ et $\ln\big(1+\frac{x^2}{n}\big)\sim\frac{x^2}{n}$, donc $\ln f_n(x)\to-\frac{x^2}{2}$ et $f_n(x)\to e^{-x^2/2}$.
\item \emph{Domination} : $0\le f_n(x)\le\frac{2}{1+x^2}$, fonction intégrable sur $\mathbb{R}$ (d'intégrale $2\pi$), indépendante de $n$. En particulier $a_n$ existe.
\end{itemize}
D'après le théorème de convergence dominée :
\[
\lim_{n\to+\infty}a_n=\int_{-\infty}^{+\infty}e^{-x^2/2}dx=2I=\boxed{\sqrt{2\pi}\approx2{,}507}
\]
(la fonction est paire, donc l'intégrale sur $\mathbb{R}$ vaut $2I$).
\remarque{La majoration $a_n\le\int\frac{2}{1+x^2}dx=2\pi$ donne seulement une borne, pas la limite : la valeur $2\pi$ n'est pas la limite de $a_n$.}
