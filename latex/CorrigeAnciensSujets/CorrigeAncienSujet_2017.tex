\textbf{Exercice 1} \quad $M=\begin{pmatrix}11&-5&5\\-5&3&-3\\5&-3&3\end{pmatrix}$.

\textbf{1. Polynôme caractéristique et spectre.}

Avec $C_3\leftarrow C_3+C_2$ puis $L_2\leftarrow L_2-L_3$ :
\[
P_M(x)=\begin{vmatrix}11-x&-5&5\\-5&3-x&-3\\5&-3&3-x\end{vmatrix}
=\begin{vmatrix}11-x&-5&0\\-5&3-x&-x\\5&-3&-x\end{vmatrix}
=\begin{vmatrix}11-x&-5&0\\-10&6-x&0\\5&-3&-x\end{vmatrix}.
\]
En développant suivant la troisième colonne :
\[
P_M(x)=-x\big[(11-x)(6-x)-50\big]=-x(x^2-17x+16).
\]
Le trinôme $x^2-17x+16$ a pour discriminant $289-64=225=15^2$ et pour racines $\frac{17\pm15}{2}$, soit $1$ et $16$. Donc
\[
\boxed{P_M(x)=-x(x-1)(x-16)},\qquad\boxed{\operatorname{Sp}(M)=\{0\,;\,1\,;\,16\}}.
\]

\textbf{2. Sous-espaces propres.}

\underline{$\lambda=0$ :} $MX=0\iff\begin{cases}11x-5y+5z=0\\5x-3y+3z=0\end{cases}$ (la deuxième ligne de $M$ est l'opposée de la troisième). En posant $w=y-z$ : $11x=5w$ et $5x=3w$, d'où $x=0$ et $w=0$. Donc $E_0=\operatorname{Vect}(u_1)$, $u_1=(0,1,1)$.

\underline{$\lambda=1$ :} $(M-I)X=0\iff\begin{cases}10x-5y+5z=0\\-5x+2y-3z=0\\5x-3y+2z=0\end{cases}$. La somme des deux dernières donne $-y-z=0$, soit $z=-y$ ; la première donne alors $10x-10y=0$, $x=y$. Donc $E_1=\operatorname{Vect}(u_2)$, $u_2=(1,1,-1)$.

\underline{$\lambda=16$ :} $(M-16I)X=0\iff\begin{cases}-5x-5y+5z=0\\-5x-13y-3z=0\\5x-3y-13z=0\end{cases}$. La somme des deux dernières donne $-16y-16z=0$, soit $z=-y$ ; la première donne $-5x-10y=0$, $x=-2y$. Donc $E_{16}=\operatorname{Vect}(u_3)$, $u_3=(2,-1,1)$.

(Vérifications : $Mu_1=0$, $Mu_2=u_2$, $Mu_3=(32,-16,16)=16u_3$.)

\textbf{3. Diagonalisation.} $M$ est une matrice d'ordre 3 ayant 3 valeurs propres distinctes : elle est \textbf{diagonalisable}. (On remarque aussi que $M$ est symétrique réelle, et que $u_1,u_2,u_3$ sont deux à deux orthogonaux.) Dans la base $\mathcal{B}=(u_1,u_2,u_3)$ :
\[
D=\begin{pmatrix}0&0&0\\0&1&0\\0&0&16\end{pmatrix},\qquad P=\begin{pmatrix}0&1&2\\1&1&-1\\1&-1&1\end{pmatrix},\qquad M=PDP^{-1}.
\]
\underline{Calcul de $P^{-1}$.} $\det P=0-1\cdot(1+1)+2\cdot(-1-1)=-6$. Par la comatrice (ou en utilisant l'orthogonalité des colonnes : $P^{-1}$ a pour lignes $\frac{{}^tu_i}{\|u_i\|^2}$ avec $\|u_1\|^2=2$, $\|u_2\|^2=3$, $\|u_3\|^2=6$) :
\[
P^{-1}=\frac16\begin{pmatrix}0&3&3\\2&2&-2\\2&-1&1\end{pmatrix}.
\]

\textbf{4. Calcul de $M^n$.}

Pour $n=0$, $M^0=I_3$. Pour $n\ge1$, $M^n=PD^nP^{-1}$ avec $D^n=\operatorname{diag}(0,1,16^n)$ (attention : $0^n=0$ seulement pour $n\ge1$). On obtient
\[
PD^n=\begin{pmatrix}0&1&2\cdot16^n\\0&1&-16^n\\0&-1&16^n\end{pmatrix},\qquad
\boxed{M^n=\frac16\begin{pmatrix}2+4\cdot16^n&2-2\cdot16^n&-2+2\cdot16^n\\2-2\cdot16^n&2+16^n&-2-16^n\\-2+2\cdot16^n&-2-16^n&2+16^n\end{pmatrix}}\quad(n\ge1).
\]
Vérification pour $n=1$ : $\frac16(2+64)=11$, $\frac16(2-32)=-5$, $\frac16(2+16)=3$ : on retrouve $M$.

\medskip\textbf{Exercice 2} \quad $(E)$ : $2y''-y'-y=-7-3x+(12x+5)e^x$.

\textbf{1. Équation homogène.} $2r^2-r-1=0$ : $\Delta=1+8=9$, $r_1=\frac{1-3}{4}=-\frac12$, $r_2=\frac{1+3}4=1$. Donc
\[
y_H(x)=Ae^{-x/2}+Be^{x},\qquad A,B\in\mathbb{R}.
\]

\textbf{2. Solution particulière de $2y''-y'-y=-7-3x$.} Avec $y_1=mx+p$ : $y_1'=m$, $y_1''=0$, donc $2y_1''-y_1'-y_1=-mx-(m+p)$. Par identification : $m=3$ et $m+p=7$, soit $p=4$ :
\[
y_1(x)=3x+4 .
\]

\textbf{3. Solution particulière de $2y''-y'-y=(12x+5)e^x$.} Le nombre $1$ étant racine simple de l'équation caractéristique, on cherche $y_2=x(ax+b)e^x=(ax^2+bx)e^x$ :
\[
y_2'=\big(ax^2+(2a+b)x+b\big)e^x,\qquad y_2''=\big(ax^2+(4a+b)x+2a+2b\big)e^x .
\]
Alors
\[
2y_2''-y_2'-y_2=\big[(2a-a-a)x^2+\big(2(4a+b)-(2a+b)-b\big)x+\big(2(2a+2b)-b\big)\big]e^x=\big(6ax+4a+3b\big)e^x .
\]
Par identification avec $(12x+5)e^x$ : $6a=12$ et $4a+3b=5$, d'où $a=2$ et $b=-1$ :
\[
\boxed{y_2(x)=x(2x-1)e^x}.
\]
(Vérification : $y_2=(2x^2-x)e^x$, $y_2'=(2x^2+3x-1)e^x$, $y_2''=(2x^2+7x+2)e^x$ et $2y_2''-y_2'-y_2=(12x+5)e^x$.)

\textbf{4. Solution particulière de $(E)$.} Par le principe de superposition :
\[
y_p(x)=y_1(x)+y_2(x)=3x+4+(2x^2-x)e^x .
\]

\textbf{5. Solution générale.}
\[
\boxed{y_G(x)=3x+4+(2x^2-x)e^x+Ae^{-x/2}+Be^x,\qquad A,B\in\mathbb{R}}.
\]

\textbf{6. Solution $f$ telle que la tangente en $A(0,2)$ soit parallèle à $(D)$.}

$(D)$ : $y=9x-9$ a pour coefficient directeur $9$. Les conditions sont donc $f(0)=2$ (la courbe passe par $A$) et $f'(0)=9$ (tangente parallèle à $(D)$).
\begin{itemize}
\item $f(0)=4+A+B=2$, soit $A+B=-2$ ;
\item $f'(x)=3+(2x^2+3x-1)e^x-\frac A2e^{-x/2}+Be^x$, donc $f'(0)=3-1-\frac A2+B=9$, soit $-\frac A2+B=7$.
\end{itemize}
En soustrayant : $\frac32A=-9$, $A=-6$, puis $B=-2-A=4$.
\[
\boxed{f(x)=3x+4+(2x^2-x)e^x-6e^{-x/2}+4e^x}.
\]

\medskip\textbf{Exercice 3} \quad $f(x)=\dfrac2\pi\displaystyle\int_0^{\pi/2}\sqrt{1+x\cos^2t}\,dt$, $x\ge-1$.

Pour $x\ge-1$ et $t\in[0,\frac\pi2]$, $1+x\cos^2t\ge1-\cos^2t=\sin^2t\ge0$ : la racine est bien définie et la fonction $t\mapsto\sqrt{1+x\cos^2t}$ est continue sur le segment $[0,\frac\pi2]$, donc $f(x)$ existe.

\textbf{1. Valeurs particulières.}
\[
f(-1)=\frac2\pi\int_0^{\pi/2}\sqrt{\sin^2t}\,dt=\frac2\pi\int_0^{\pi/2}\sin t\,dt=\frac2\pi\Big[-\cos t\Big]_0^{\pi/2}=\boxed{\frac2\pi},
\qquad f(0)=\frac2\pi\int_0^{\pi/2}dt=\boxed{1}
\]
(on a $\sqrt{\sin^2t}=|\sin t|=\sin t$ sur $[0,\frac\pi2]$).

\textbf{2. Relation $f(x)=\sqrt{1+x}\,f\big(\frac{-x}{1+x}\big)$ pour $x>-1$.}

Remarquons d'abord que $y=\frac{-x}{1+x}$ vérifie $1+y=\frac1{1+x}>0$, donc $y>-1$ et $f(y)$ est défini. Alors
\[
\sqrt{1+x}\,\sqrt{1+y\cos^2t}=\sqrt{(1+x)-x\cos^2t}=\sqrt{1+x\sin^2t},
\]
d'où $\sqrt{1+x}\,f(y)=\dfrac2\pi\displaystyle\int_0^{\pi/2}\sqrt{1+x\sin^2t}\,dt$. Avec le changement de variable $u=\frac\pi2-t$ ($\sin t=\cos u$) :
\[
\sqrt{1+x}\,f\Big(\frac{-x}{1+x}\Big)=\frac2\pi\int_0^{\pi/2}\sqrt{1+x\cos^2u}\,du=f(x).
\]

\textbf{3. Encadrement $0\le f(x)-f(x_0)\le\frac{x-x_0}{4}$ pour $0\le x_0\le x$.}

\remarque{L'énoncé original indiquait $-1\le x_0\le x$ ; l'inégalité de droite est alors fausse (par exemple $f(0)-f(-\frac12)\approx0{,}140>\frac18$). Elle est vraie pour $x_0\ge0$, et on la démontre dans ce cadre.}

Pour $x_0\le x$ et $t\in[0,\frac\pi2]$, en multipliant par la quantité conjuguée :
\[
\sqrt{1+x\cos^2t}-\sqrt{1+x_0\cos^2t}=\frac{(x-x_0)\cos^2t}{\sqrt{1+x\cos^2t}+\sqrt{1+x_0\cos^2t}}\ \ge0
\]
(si le dénominateur est nul, la différence est nulle ; cela n'arrive qu'en $t=0$ pour $x=x_0=-1$). En intégrant, $f(x)-f(x_0)\ge0$ : \textbf{$f$ est croissante sur $[-1,+\infty[$}.

Si de plus $0\le x_0\le x$, chaque racine est $\ge1$, donc le dénominateur est $\ge2$ et
\[
0\le f(x)-f(x_0)\le\frac2\pi\int_0^{\pi/2}\frac{(x-x_0)\cos^2t}{2}\,dt=\frac{x-x_0}{\pi}\cdot\frac\pi4=\frac{x-x_0}{4},
\]
car $\displaystyle\int_0^{\pi/2}\cos^2t\,dt=\int_0^{\pi/2}\frac{1+\cos2t}{2}dt=\frac\pi4$.

(Plus généralement, pour $-1<x_0\le x$, le dénominateur est $\ge2\sqrt{1+x_0}$, ce qui donne $0\le f(x)-f(x_0)\le\dfrac{x-x_0}{4\sqrt{1+x_0}}$.)

\textbf{4. Encadrement $0\le f(x)-f(-1)\le\frac12\sqrt{1+x}$ pour $-1<x<0$.}

Posons $\varepsilon=1+x\in]0,1[$. Comme $1+x\cos^2t=\sin^2t+\varepsilon\cos^2t$ :
\[
f(x)-f(-1)=\frac2\pi\int_0^{\pi/2}\Big(\sqrt{\sin^2t+\varepsilon\cos^2t}-\sin t\Big)dt=\frac2\pi\int_0^{\pi/2}\frac{\varepsilon\cos^2t}{\sqrt{\sin^2t+\varepsilon\cos^2t}+\sin t}\,dt\ \ge0 .
\]
Minorons le dénominateur. On a $\sqrt{\sin^2t+\varepsilon\cos^2t}\ge\sqrt\varepsilon\cos t$ et, comme $0<\sqrt\varepsilon<1$, $\sin t\ge\sqrt\varepsilon\sin t$. Donc
\[
\sqrt{\sin^2t+\varepsilon\cos^2t}+\sin t\ge\sqrt\varepsilon(\cos t+\sin t)\ge\sqrt\varepsilon,
\]
car $(\cos t+\sin t)^2=1+\sin2t\ge1$ et $\cos t+\sin t\ge0$ sur $[0,\frac\pi2]$. Il vient
\[
0\le f(x)-f(-1)\le\frac2\pi\int_0^{\pi/2}\frac{\varepsilon\cos^2t}{\sqrt\varepsilon}\,dt=\frac2\pi\sqrt\varepsilon\cdot\frac\pi4=\boxed{\frac12\sqrt{1+x}}.
\]

\textbf{5. Continuité de $f$.}

\underline{En $-1$ (à droite).} D'après la question 4, $0\le f(x)-f(-1)\le\frac12\sqrt{1+x}\xrightarrow[x\to-1^+]{}0$, donc $\lim_{-1^+}f=f(-1)$.

\underline{En $x_0>-1$.} D'après la remarque de la question 3, pour $-1<a\le x_0,x$ :
\[
|f(x)-f(x_0)|\le\frac{|x-x_0|}{4\sqrt{1+a}},
\]
donc $f$ est lipschitzienne sur $[a,+\infty[$ pour tout $a>-1$, a fortiori continue.

\textbf{Conclusion :} $f$ est continue (et croissante) sur $[-1,+\infty[$. (On peut aussi invoquer le théorème de continuité des intégrales dépendant d'un paramètre : $(x,t)\mapsto\sqrt{1+x\cos^2t}$ est continue sur $[-1,+\infty[\times[0,\frac\pi2]$ et l'on intègre sur un segment.)
