\textbf{Exercice 1} \quad $M=\begin{pmatrix}11&-5&5\\-5&3&-3\\5&-3&3\end{pmatrix}$.

\textbf{1. Polynôme caractéristique et spectre.}

Avec $C_3\leftarrow C_3+C_2$ puis $L_2\leftarrow L_2-L_3$ :
\[
P_M(x)=\begin{vmatrix}11-x&-5&5\\-5&3-x&-3\\5&-3&3-x\end{vmatrix}
=\begin{vmatrix}11-x&-5&0\\-5&3-x&-x\\5&-3&-x\end{vmatrix}
=\begin{vmatrix}11-x&-5&0\\-10&6-x&0\\5&-3&-x\end{vmatrix}.
\]
En développant suivant la troisième colonne :
\[
P_M(x)=-x\big[(11-x)(6-x)-50\big]=-x(x^2-17x+16).
\]
Le trinôme $x^2-17x+16$ a pour discriminant $289-64=225=15^2$ et pour racines $\frac{17\pm15}{2}$, soit $1$ et $16$. Donc
\[
\boxed{P_M(x)=-x(x-1)(x-16)},\qquad\boxed{\operatorname{Sp}(M)=\{0\,;\,1\,;\,16\}}.
\]

\textbf{2. Sous-espaces propres.}

\underline{$\lambda=0$ :} $MX=0\iff\begin{cases}11x-5y+5z=0\\5x-3y+3z=0\end{cases}$ (la deuxième ligne de $M$ est l'opposée de la troisième). En posant $w=y-z$ : $11x=5w$ et $5x=3w$, d'où $x=0$ et $w=0$. Donc $E_0=\operatorname{Vect}(u_1)$, $u_1=(0,1,1)$.

\underline{$\lambda=1$ :} $(M-I)X=0\iff\begin{cases}10x-5y+5z=0\\-5x+2y-3z=0\\5x-3y+2z=0\end{cases}$. La somme des deux dernières donne $-y-z=0$, soit $z=-y$ ; la première donne alors $10x-10y=0$, $x=y$. Donc $E_1=\operatorname{Vect}(u_2)$, $u_2=(1,1,-1)$.

\underline{$\lambda=16$ :} $(M-16I)X=0\iff\begin{cases}-5x-5y+5z=0\\-5x-13y-3z=0\\5x-3y-13z=0\end{cases}$. La somme des deux dernières donne $-16y-16z=0$, soit $z=-y$ ; la première donne $-5x-10y=0$, $x=-2y$. Donc $E_{16}=\operatorname{Vect}(u_3)$, $u_3=(2,-1,1)$.

(Vérifications : $Mu_1=0$, $Mu_2=u_2$, $Mu_3=(32,-16,16)=16u_3$.)

\textbf{3. Diagonalisation.} $M$ est une matrice d'ordre 3 ayant 3 valeurs propres distinctes : elle est \textbf{diagonalisable}. (On remarque aussi que $M$ est symétrique réelle, et que $u_1,u_2,u_3$ sont deux à deux orthogonaux.) Dans la base $\mathcal{B}=(u_1,u_2,u_3)$ :
\[
D=\begin{pmatrix}0&0&0\\0&1&0\\0&0&16\end{pmatrix},\qquad P=\begin{pmatrix}0&1&2\\1&1&-1\\1&-1&1\end{pmatrix},\qquad M=PDP^{-1}.
\]
\underline{Calcul de $P^{-1}$.} $\det P=0-1\cdot(1+1)+2\cdot(-1-1)=-6$. Par la comatrice (ou en utilisant l'orthogonalité des colonnes : $P^{-1}$ a pour lignes $\frac{{}^tu_i}{\|u_i\|^2}$ avec $\|u_1\|^2=2$, $\|u_2\|^2=3$, $\|u_3\|^2=6$) :
\[
P^{-1}=\frac16\begin{pmatrix}0&3&3\\2&2&-2\\2&-1&1\end{pmatrix}.
\]

\textbf{4. Calcul de $M^n$.}

Pour $n=0$, $M^0=I_3$. Pour $n\ge1$, $M^n=PD^nP^{-1}$ avec $D^n=\operatorname{diag}(0,1,16^n)$ (attention : $0^n=0$ seulement pour $n\ge1$). On obtient
\[
PD^n=\begin{pmatrix}0&1&2\cdot16^n\\0&1&-16^n\\0&-1&16^n\end{pmatrix},
\]
\[
\boxed{M^n=\frac16\begin{pmatrix}2+4\cdot16^n&2-2\cdot16^n&-2+2\cdot16^n\\2-2\cdot16^n&2+16^n&-2-16^n\\-2+2\cdot16^n&-2-16^n&2+16^n\end{pmatrix}}\quad(n\ge1).
\]
Vérification pour $n=1$ : $\frac16(2+64)=11$, $\frac16(2-32)=-5$, $\frac16(2+16)=3$ : on retrouve $M$.

\medskip\textbf{Exercice 2} \quad $h$ est $2\pi$-périodique, impaire, et $h(x)=x(\pi-x)$ pour $x\in[0,\pi]$.

\textbf{1. Représentation graphique sur $[-2\pi,2\pi]$.}

\underline{Sur $[0,\pi]$} : $h'(x)=\pi-2x$ ; $h$ croît sur $[0,\frac\pi2]$, décroît sur $[\frac\pi2,\pi]$, avec $h(0)=h(\pi)=0$ et un maximum $h(\frac\pi2)=\frac{\pi^2}{4}\approx2{,}47$ ; la courbe est un arc de parabole.

\underline{Sur $[-\pi,0]$} : $h$ étant impaire, $h(x)=-h(-x)=-(-x)(\pi+x)=x(\pi+x)$ : la courbe est symétrique de la précédente par rapport à l'\textbf{origine} $O$.

\underline{Sur $\mathbb{R}$} : on complète par périodicité (translations de vecteur $2\pi\,\vec\imath$). $h$ est continue sur $\mathbb{R}$ (les raccords en $0$ et en $\pm\pi$ se font à la valeur $0$).

\textbf{2. Coefficients de Fourier.}

$h$ est impaire, donc $a_n=\frac1\pi\int_{-\pi}^{\pi}h(x)\cos(nx)\,dx=0$ pour tout $n\ge0$, et
\[
b_n=\frac1\pi\int_{-\pi}^{\pi}h(x)\sin(nx)\,dx=\frac2\pi\int_0^\pi x(\pi-x)\sin(nx)\,dx\qquad(n\ge1).
\]
On intègre deux fois par parties ($u=x(\pi-x)$, $u'=\pi-2x$, $u''=-2$) :
\[
\begin{split}
\int_0^\pi x(\pi-x)\sin(nx)\,dx&=\Big[-x(\pi-x)\frac{\cos nx}{n}\Big]_0^\pi+\frac1n\int_0^\pi(\pi-2x)\cos(nx)\,dx\\
&=\frac1n\int_0^\pi(\pi-2x)\cos(nx)\,dx,
\end{split}
\]
car le crochet est nul (le polynôme s'annule en $0$ et en $\pi$). Puis
\[
\begin{split}
\int_0^\pi(\pi-2x)\cos(nx)\,dx&=\Big[(\pi-2x)\frac{\sin nx}{n}\Big]_0^\pi+\frac2n\int_0^\pi\sin(nx)\,dx\\
&=0+\frac2n\Big[-\frac{\cos nx}{n}\Big]_0^\pi=\frac{2\big(1-(-1)^n\big)}{n^2}.
\end{split}
\]
Donc
\[
\boxed{b_n=\frac{4\big(1-(-1)^n\big)}{\pi n^3}}=\begin{cases}\dfrac{8}{\pi n^3}&\text{si } n \text{ est impair,}\\[2mm]0&\text{si } n \text{ est pair.}\end{cases}
\]
(Contrôle : $b_1=\frac8\pi>0$, cohérent avec $h>0$ sur $]0,\pi[$.)

\textbf{3. Série de Fourier.} $h$ est continue sur $\mathbb{R}$ et de classe $C^1$ par morceaux. D'après le théorème de Dirichlet, sa série de Fourier converge vers $h(x)$ en tout point $x$ :
\[
\boxed{h(x)=\frac8\pi\sum_{k=0}^{+\infty}\frac{\sin\big((2k+1)x\big)}{(2k+1)^3}},\qquad x\in\mathbb{R}.
\]

\textbf{4. Sommes $S_1$ et $S_2$.}

\underline{$S_1$ : valeur en $x=\frac\pi2$.} $\sin\big((2k+1)\frac\pi2\big)=(-1)^k$ et $h(\frac\pi2)=\frac{\pi^2}{4}$, donc
\[
\frac{\pi^2}{4}=\frac8\pi\sum_{k=0}^{+\infty}\frac{(-1)^k}{(2k+1)^3},\qquad\boxed{S_1=\sum_{n=0}^{+\infty}\frac{(-1)^n}{(2n+1)^3}=\frac{\pi^3}{32}\approx0{,}9689}.
\]

\underline{$S_2$ : formule de Parseval.} Pour $h$ continue par morceaux et $2\pi$-périodique :
\[
\frac1\pi\int_{-\pi}^{\pi}h(x)^2\,dx=\frac{a_0^2}{2}+\sum_{n\ge1}\big(a_n^2+b_n^2\big).
\]
Ici $h^2$ est paire, et
\[
\int_0^\pi x^2(\pi-x)^2dx=\int_0^\pi\big(\pi^2x^2-2\pi x^3+x^4\big)dx=\frac{\pi^5}{3}-\frac{\pi^5}{2}+\frac{\pi^5}{5}=\frac{\pi^5}{30},
\]
donc $\frac1\pi\int_{-\pi}^\pi h^2=\frac2\pi\cdot\frac{\pi^5}{30}=\frac{\pi^4}{15}$. D'autre part $\sum b_n^2=\frac{64}{\pi^2}\sum_{k\ge0}\frac1{(2k+1)^6}$. D'où
\[
\frac{\pi^4}{15}=\frac{64}{\pi^2}S_2,\qquad\boxed{S_2=\sum_{n=0}^{+\infty}\frac{1}{(2n+1)^6}=\frac{\pi^6}{960}\approx1{,}0014}.
\]

\medskip\textbf{Exercice 3}

Notons $P$ l'évènement « la machine est tombée en panne en cinq ans » et $H$ l'évènement « la machine est hors d'usage ». Les données sont :
\[
P(P)=0{,}3,\qquad P(\overline P)=0{,}7,\qquad P_P(H)=0{,}75,\qquad P_{\overline P}(H)=0{,}4 .
\]
(Arbre : première branche $P$ / $\overline P$, puis $H$ / $\overline H$.)

\textbf{1. Probabilité d'être hors d'usage.} D'après la formule des probabilités totales (système complet $\{P,\overline P\}$) :
\[
P(H)=P(P)P_P(H)+P(\overline P)P_{\overline P}(H)=0{,}3\times0{,}75+0{,}7\times0{,}4=0{,}225+0{,}28=\boxed{0{,}505}.
\]

\textbf{2. Probabilité de n'avoir jamais eu de panne sachant que la machine est hors d'usage.} Par la formule de Bayes :
\[
P_H(\overline P)=\frac{P(\overline P\cap H)}{P(H)}=\frac{0{,}28}{0{,}505}=\boxed{\frac{56}{101}\approx0{,}554}.
\]

\textbf{3. Loi de $X$.} On choisit 10 machines ; chacune tombe en panne en cinq ans avec la probabilité $p=0{,}3$, indépendamment des autres (épreuves de Bernoulli indépendantes et identiques, le parc étant supposé grand). $X$ compte le nombre de pannes : \textbf{$X$ suit la loi binomiale $\mathcal{B}(10\,;\,0{,}3)$} :
\[
P(X=k)=C_{10}^k\,(0{,}3)^k\,(0{,}7)^{10-k},\qquad k\in\{0,1,\dots,10\}.
\]
Par exemple $P(X=0)=0{,}7^{10}\approx0{,}028$ et $P(X=3)=120\times0{,}027\times0{,}7^7\approx0{,}267$ (valeur la plus probable).
\[
E(X)=np=10\times0{,}3=\boxed{3},\qquad V(X)=np(1-p)=10\times0{,}3\times0{,}7=\boxed{2{,}1},
\]
\[
\sigma(X)=\sqrt{2{,}1}\approx\boxed{1{,}45}.
\]
