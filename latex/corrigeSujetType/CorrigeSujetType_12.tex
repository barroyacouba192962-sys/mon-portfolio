\textbf{Exercice 1}

\textbf{1. Volume de $D=\{0\le x\le1,\ 0\le y\le(1-\sqrt x)^2,\ z^2\le2xy\}$.}

\underline{Bornes.} $x$ varie dans $[0,1]$ ; pour $x$ fixé, $y\in[0,(1-\sqrt x)^2]$ ; pour $(x,y)$ fixé ($xy\ge0$), $z^2\le2xy\iff-\sqrt{2xy}\le z\le\sqrt{2xy}$.

\underline{Calcul.} Par le théorème de Fubini :
\[
V=\int_0^1\int_0^{(1-\sqrt x)^2}\Big(\int_{-\sqrt{2xy}}^{\sqrt{2xy}}dz\Big)dy\,dx=\int_0^1\int_0^{(1-\sqrt x)^2}2\sqrt{2x}\,\sqrt y\,dy\,dx .
\]
Intégrale en $y$ (avec $\int\sqrt y\,dy=\frac23y^{3/2}$ et $\big((1-\sqrt x)^2\big)^{3/2}=(1-\sqrt x)^3$ car $1-\sqrt x\ge0$) :
\[
\int_0^{(1-\sqrt x)^2}2\sqrt{2x}\sqrt y\,dy=2\sqrt{2x}\cdot\frac23(1-\sqrt x)^3=\frac{4\sqrt2}{3}\sqrt x\,(1-\sqrt x)^3 .
\]
Intégrale en $x$ : on pose $u=\sqrt x$ ($x=u^2$, $dx=2u\,du$, $u\in[0,1]$) :
\[
V=\frac{4\sqrt2}{3}\int_0^1u(1-u)^3\,2u\,du=\frac{8\sqrt2}{3}\int_0^1u^2(1-u)^3\,du .
\]
En développant $u^2(1-u)^3=u^2-3u^3+3u^4-u^5$ :
\[
\int_0^1u^2(1-u)^3du=\frac13-\frac34+\frac35-\frac16=\frac{20-45+36-10}{60}=\frac1{60}.
\]
\[
\boxed{V=\frac{8\sqrt2}{3}\times\frac1{60}=\frac{2\sqrt2}{45}\approx0{,}0629}.
\]

\textbf{2. $T=\displaystyle\iint_D(x+y)^2e^{x^2-y^2}dx\,dy$, $D=\{x+y\le1,\ x\ge0,\ y\ge0\}$.}

\underline{Changement de variables.} $x=\frac{u+v}{2}$, $y=\frac{u-v}{2}$, c'est-à-dire $u=x+y$, $v=x-y$. Le jacobien vaut
\[
\frac{D(x,y)}{D(u,v)}=\begin{vmatrix}\frac12&\frac12\\[1mm]\frac12&-\frac12\end{vmatrix}=-\frac12,\qquad\text{donc}\qquad dx\,dy=\frac12\,du\,dv .
\]
\underline{Nouveau domaine.} $x+y\le1\iff u\le1$ ; $x\ge0\iff v\ge-u$ ; $y\ge0\iff v\le u$. Donc $\Delta=\{0\le u\le1,\ -u\le v\le u\}$.

\underline{Intégrande.} $(x+y)^2=u^2$ et $x^2-y^2=(x+y)(x-y)=uv$. D'où
\[
T=\frac12\int_0^1u^2\Big(\int_{-u}^ue^{uv}dv\Big)du .
\]
Pour $u>0$ : $\displaystyle\int_{-u}^ue^{uv}dv=\Big[\frac{e^{uv}}{u}\Big]_{-u}^u=\frac{e^{u^2}-e^{-u^2}}{u}$. Donc
\[
T=\frac12\int_0^1u\big(e^{u^2}-e^{-u^2}\big)du\underset{w=u^2}{=}\frac14\int_0^1\big(e^w-e^{-w}\big)dw=\frac14\Big[e^w+e^{-w}\Big]_0^1,
\]
\[
\boxed{T=\frac14\Big(e+\frac1e-2\Big)=\frac{(e-1)^2}{4e}\approx0{,}271}.
\]

\textbf{3. $T=\displaystyle\iiint_Dx^\alpha y^\beta z\,dx\,dy\,dz$, $D=\{0\le x\le1,\ 0\le y\le1,\ 0\le z\le xy\}$.}

La fonction est continue et positive, on applique Fubini :
\[
\int_0^{xy}z\,dz=\frac{(xy)^2}{2},\qquad T=\frac12\int_0^1\int_0^1x^{\alpha+2}y^{\beta+2}\,dy\,dx=\frac12\Big(\int_0^1x^{\alpha+2}dx\Big)\Big(\int_0^1y^{\beta+2}dy\Big),
\]
\[
\boxed{T=\frac{1}{2(\alpha+3)(\beta+3)}}.
\]

\medskip\textbf{Exercice 2}

$E$ est de dimension $n$, $f\in\mathcal{L}(E)$, et on note $f^0=\mathrm{Id}_E$, $f^k=f\circ\dots\circ f$.

\textbf{1.(a) Si $f^{p-1}(x)\neq0$, la famille $(x,f(x),\dots,f^{p-1}(x))$ est libre.}

\emph{(L'énoncé original écrivait $f^{p-1}(x)=0$ ; il faut lire $f^{p-1}(x)\neq0$, sinon le résultat est faux.)}

Remarquons d'abord que $f^m=0$ pour tout $m\ge p$, car $f^m=f^{m-p}\circ f^p=0$.

Soient $\lambda_0,\dots,\lambda_{p-1}$ des réels tels que
\[
\lambda_0x+\lambda_1f(x)+\dots+\lambda_{p-1}f^{p-1}(x)=0\qquad(\ast)
\]
Supposons, par l'absurde, que les $\lambda_i$ ne soient pas tous nuls, et notons $k$ le plus petit indice tel que $\lambda_k\neq0$. Alors $(\ast)$ s'écrit $\sum_{i=k}^{p-1}\lambda_if^i(x)=0$. Appliquons $f^{p-1-k}$ (linéaire) :
\[
\sum_{i=k}^{p-1}\lambda_if^{\,p-1-k+i}(x)=0 .
\]
Pour $i\ge k+1$, l'exposant $p-1-k+i\ge p$, donc $f^{p-1-k+i}(x)=0$. Il reste $\lambda_kf^{p-1}(x)=0$, et comme $f^{p-1}(x)\neq0$, $\lambda_k=0$ : contradiction. Donc tous les $\lambda_i$ sont nuls et la famille est libre.

(On peut aussi présenter ce raisonnement comme une récurrence : en composant $(\ast)$ par $f^{p-1}$ on obtient $\lambda_0=0$, puis par $f^{p-2}$ on obtient $\lambda_1=0$, etc.)

\textbf{(b) $p\le n$ et $f^n=0$.}

Comme $p$ est l'indice de nilpotence, $f^{p-1}\neq0$ : il existe $x\in E$ tel que $f^{p-1}(x)\neq0$. D'après (a), la famille $(x,\dots,f^{p-1}(x))$ est libre et possède $p$ vecteurs. Dans un espace de dimension $n$, une famille libre a au plus $n$ éléments : $p\le n$. Alors $f^n=f^{n-p}\circ f^p=0$.

\textbf{(c) Cas $p=n$.} Comme $f^{n-1}\neq0$, il existe $x_0\in E$ tel que $f^{n-1}(x_0)\neq0$. D'après (a), la famille $(x_0,f(x_0),\dots,f^{n-1}(x_0))$ est libre ; elle a $n=\dim E$ éléments, c'est donc une base de $E$. (Dans cette base, la matrice de $f$ a des $1$ juste sous la diagonale et des $0$ ailleurs.)

\textbf{2. Si pour tout $x$ il existe $p_x$ avec $f^{p_x}(x)=0$, alors $f^n=0$.}

Soit $(e_1,\dots,e_n)$ une base de $E$. Par hypothèse, pour chaque $i$ il existe $p_i$ tel que $f^{p_i}(e_i)=0$. Posons $m=\max(p_1,\dots,p_n)$ ; alors $f^m(e_i)=f^{m-p_i}\big(f^{p_i}(e_i)\big)=0$ pour tout $i$. Pour $x=\sum x_ie_i\in E$, par linéarité, $f^m(x)=\sum x_if^m(e_i)=0$. Donc $f^m=0$ : $f$ est nilpotent, et d'après 1.(b), $f^n=0$.

\remarque{Il est indispensable de passer par une base : l'hypothèse ne donne qu'un exposant $p_x$ \emph{dépendant de $x$}, et il faut un exposant commun à tous les vecteurs.}

\medskip\textbf{Exercice 3}

On note $P_k$ (resp. $F_k$) l'évènement « pile (resp. face) au $k$-ième lancer » ; $P(P_k)=\frac23$, $P(F_k)=\frac13$, lancers indépendants.

\textbf{(1) Les évènements $(X=2)$, $(X=3)$, $(X=4)$.}
\begin{itemize}
\item $(X=2)=P_1\cap P_2$, donc $p_2=\frac23\times\frac23=\boxed{\frac49}$.
\item $(X=3)=F_1\cap P_2\cap P_3$ : les lancers 2 et 3 donnent pile, et le lancer 1 doit donner face (sinon on aurait déjà $X=2$). Donc $p_3=\frac13\cdot\frac49=\boxed{\frac4{27}}$.
\item $(X=4)=(P_1\cap F_2\cap P_3\cap P_4)\cup(F_1\cap F_2\cap P_3\cap P_4)$ : les lancers 3 et 4 donnent pile, le lancer 2 donne face, le lancer 1 est quelconque. Donc $p_4=1\times\frac13\times\frac49=\boxed{\frac4{27}}$.
\end{itemize}
(De plus $p_1=0$, car il faut au moins deux lancers.)

\textbf{(2) Relation de récurrence pour $n\ge4$.}

On utilise la formule des probabilités totales avec le système complet $\{F_1,\ P_1\cap F_2,\ P_1\cap P_2\}$ :
\begin{itemize}
\item si $P_1\cap P_2$ est réalisé, $X=2\neq n$ : contribution nulle ;
\item si $F_1$ est réalisé, tout recommence à partir du lancer 2 : il faut obtenir les deux piles consécutifs pour la première fois en $n-1$ lancers supplémentaires, avec la probabilité $p_{n-1}$ (indépendance des lancers) ;
\item si $P_1\cap F_2$ est réalisé, tout recommence à partir du lancer 3 : probabilité $p_{n-2}$.
\end{itemize}
D'où, pour $n\ge4$ :
\[
p_n=\frac13\,p_{n-1}+\frac23\cdot\frac13\,p_{n-2}=\frac29\,p_{n-2}+\frac13\,p_{n-1}.
\]
Vérification pour $n=4$ : $\frac29\cdot\frac49+\frac13\cdot\frac4{27}=\frac{8}{81}+\frac{4}{81}=\frac{12}{81}=\frac4{27}=p_4$.

\textbf{(3) Expression de $p_n$.}

C'est une récurrence linéaire d'ordre 2 : $p_n-\frac13p_{n-1}-\frac29p_{n-2}=0$, d'équation caractéristique
\[
r^2-\frac13r-\frac29=0,\qquad\Delta=\frac19+\frac89=1,\qquad r_1=\frac23,\ r_2=-\frac13 .
\]
Donc il existe $A,B$ tels que $p_n=A\big(\frac23\big)^n+B\big(-\frac13\big)^n$ pour $n\ge2$. Avec $p_2=\frac49$ et $p_3=\frac4{27}$ :
\[
\begin{cases}\frac{4A}{9}+\frac{B}{9}=\frac49\\[1mm]\frac{8A}{27}-\frac{B}{27}=\frac4{27}\end{cases}\iff\begin{cases}4A+B=4\\8A-B=4\end{cases}\iff A=\frac23,\ B=\frac43 .
\]
\[
\boxed{p_n=\Big(\frac23\Big)^{n+1}+\frac43\Big(-\frac13\Big)^n}\qquad(n\ge1),
\]
la formule étant aussi valable pour $n=1$ ($\frac49-\frac49=0=p_1$). Vérification : $\sum_{n\ge1}p_n=\frac23\cdot\frac{2/3}{1-2/3}+\frac43\cdot\frac{-1/3}{1+1/3}=\frac43-\frac13=1$.

\textbf{(4) Espérance.} Pour $|q|<1$, en dérivant $\sum_{n\ge0}q^n=\frac{1}{1-q}$ puis en multipliant par $q$ :
\[
\sum_{n=0}^{+\infty}nq^n=\frac{q}{(1-q)^2}.
\]
La série $\sum np_n$ converge (combinaison de séries de ce type avec $q=\frac23$ et $q=-\frac13$), et
\begin{align*}
E(X)&=\sum_{n=1}^{+\infty}np_n=\frac23\sum_{n=1}^{+\infty}n\Big(\frac23\Big)^n+\frac43\sum_{n=1}^{+\infty}n\Big(-\frac13\Big)^n\\
&=\frac23\cdot\frac{2/3}{(1/3)^2}+\frac43\cdot\frac{-1/3}{(4/3)^2}=\frac23\times6+\frac43\times\Big(-\frac{3}{16}\Big)=4-\frac14 .
\end{align*}
\[
\boxed{E(X)=\frac{15}{4}=3{,}75}.
\]
\remarque{On retrouve la formule générale : pour une probabilité $p$ de pile, le nombre moyen de lancers pour obtenir deux piles consécutifs est $\dfrac{1+p}{p^2}$, soit ici $\dfrac{5/3}{4/9}=\dfrac{15}{4}$.}

\medskip\textbf{Exercice 4}

$f$ est continue sur $[0,a]$, $f(x)\neq-1$ et $f(x)f(a-x)=1$ pour tout $x\in[0,a]$.

\underline{Existence de $I$.} $x\mapsto\frac1{1+f(x)}$ est continue sur le segment $[0,a]$ (car $1+f(x)\neq0$) : $I$ existe. De plus $f(x)\neq0$ (car $f(x)f(a-x)=1$), et $f(a-x)=\dfrac1{f(x)}$.

\underline{Changement de variable $t=a-x$} ($dt=-dx$, $x=0\mapsto t=a$, $x=a\mapsto t=0$) :
\[
I=\int_0^a\frac{dx}{1+f(x)}=\int_a^0\frac{-dt}{1+f(a-t)}=\int_0^a\frac{dt}{1+f(a-t)}=\int_0^a\frac{dt}{1+\frac1{f(t)}}=\int_0^a\frac{f(t)}{1+f(t)}dt .
\]
\underline{Somme.} En additionnant les deux expressions de $I$ :
\[
2I=\int_0^a\frac{1+f(x)}{1+f(x)}dx=\int_0^adx=a,\qquad\boxed{I=\frac a2}.
\]
(Exemple : $f(x)=e^{x-a/2}$ vérifie les hypothèses, et on vérifie numériquement $I=\frac a2$.)
