\textbf{Exercice 1}

\textbf{1. Forme quadratique $q(x,y,z)=x(x-4y+4z)+2yz=x^2-4xy+4xz+2yz$.}

La matrice symétrique associée a pour coefficients diagonaux les coefficients des carrés et pour coefficients non diagonaux la moitié des coefficients des termes rectangles :
\[
A=\begin{pmatrix}1&-2&2\\-2&0&1\\2&1&0\end{pmatrix},\qquad q(X)={}^tXAX .
\]

\textbf{(a) Signature.}

\underline{Méthode de Gauss.} On regroupe les termes en $x$ : $x^2-4xy+4xz=(x-2y+2z)^2-(2y-2z)^2$. Donc
\[
q=(x-2y+2z)^2-4y^2+8yz-4z^2+2yz=(x-2y+2z)^2-4y^2+10yz-4z^2 .
\]
Puis on regroupe les termes en $y$ : $-4y^2+10yz=-4\big(y^2-\frac52yz\big)=-4\big(y-\frac54z\big)^2+\frac{25}{4}z^2$. Ainsi
\[
\boxed{q(x,y,z)=(x-2y+2z)^2-4\Big(y-\frac54z\Big)^2+\frac94z^2}.
\]
Les trois formes linéaires $x-2y+2z$, $y-\frac54z$, $z$ sont indépendantes (système triangulaire). On compte deux coefficients positifs et un négatif : la signature est $\boxed{(2,1)}$ ; $q$ est non dégénérée ($\operatorname{rg}q=3$).

\underline{Méthode des mineurs principaux emboîtés.}
\[
\Delta_1=1,\qquad\Delta_2=\begin{vmatrix}1&-2\\-2&0\end{vmatrix}=0-4=-4,
\]
\[
\Delta_3=\det A=1(0-1)+2(0-2)+2(-2-0)=-9 .
\]
Lorsque tous les $\Delta_k$ sont non nuls, le nombre de valeurs propres négatives de $A$ est égal au nombre de changements de signe dans la suite $(\Delta_0=1,\Delta_1,\Delta_2,\Delta_3)=(1,\,1,\,-4,\,-9)$ (méthode de Jacobi). Il y a un seul changement de signe (entre $1$ et $-4$) : une valeur propre négative et deux positives. La signature est bien $(2,1)$.

(Vérification : les valeurs propres de $A$ sont $3$, $1$ et $-3$, et le produit des coefficients de Gauss $1\times(-4)\times\frac94=-9=\det A$.)

\textbf{(b) Factorisation $QR$ de $A$} ($Q$ orthogonale, $R$ triangulaire supérieure à diagonale positive), par le procédé de Gram-Schmidt appliqué aux colonnes
\[
a_1=\begin{pmatrix}1\\-2\\2\end{pmatrix},\quad a_2=\begin{pmatrix}-2\\0\\1\end{pmatrix},\quad a_3=\begin{pmatrix}2\\1\\0\end{pmatrix}.
\]
\begin{itemize}
\item $\|a_1\|=\sqrt{1+4+4}=3$, \quad $q_1=\dfrac13\begin{pmatrix}1\\-2\\2\end{pmatrix}$, \quad $r_{11}=3$.
\item $r_{12}=\langle a_2,q_1\rangle=\frac13(-2+0+2)=0$ : $a_2$ est déjà orthogonal à $q_1$. $r_{22}=\|a_2\|=\sqrt5$, \quad $q_2=\dfrac1{\sqrt5}\begin{pmatrix}-2\\0\\1\end{pmatrix}$.
\item $r_{13}=\langle a_3,q_1\rangle=\frac13(2-2+0)=0$ et $r_{23}=\langle a_3,q_2\rangle=\frac{1}{\sqrt5}(-4+0+0)=-\frac4{\sqrt5}$. Alors
\[
b_3=a_3-r_{13}q_1-r_{23}q_2=\begin{pmatrix}2\\1\\0\end{pmatrix}+\frac45\begin{pmatrix}-2\\0\\1\end{pmatrix}=\begin{pmatrix}\frac25\\[1mm]1\\[1mm]\frac45\end{pmatrix},\qquad r_{33}=\|b_3\|=\sqrt{\frac{4+25+16}{25}}=\frac{3}{\sqrt5},
\]
\[
q_3=\frac{b_3}{\|b_3\|}=\frac{\sqrt5}{3}\cdot\frac15\begin{pmatrix}2\\5\\4\end{pmatrix}=\frac{1}{3\sqrt5}\begin{pmatrix}2\\5\\4\end{pmatrix}.
\]
\end{itemize}
D'où
\[
\boxed{Q=\begin{pmatrix}
\frac13 & -\frac2{\sqrt5} & \frac{2}{3\sqrt5}\\[1mm]
-\frac23 & 0 & \frac{5}{3\sqrt5}\\[1mm]
\frac23 & \frac1{\sqrt5} & \frac{4}{3\sqrt5}
\end{pmatrix},\qquad
R=\begin{pmatrix}
3 & 0 & 0\\
0 & \sqrt5 & -\frac4{\sqrt5}\\[1mm]
0 & 0 & \frac3{\sqrt5}
\end{pmatrix}}
\]
et $A=QR$. (Vérification : $\det R=9$, $\det Q=-1$, et $\det A=-9$ ; la troisième colonne de $QR$ vaut $-\frac4{\sqrt5}q_2+\frac3{\sqrt5}q_3=\frac45(2,0,-1)+\frac15(2,5,4)=(2,1,0)=a_3$.)

\textbf{2.(a) Nature des séries.}

\underline{$\sum\dfrac{(n+2)^n}{n^{n+2}}$.} On écrit
\[
a_n=\frac{(n+2)^n}{n^n}\cdot\frac1{n^2}=\Big(1+\frac2n\Big)^n\frac1{n^2},\qquad\Big(1+\frac2n\Big)^n=e^{n\ln(1+2/n)}\xrightarrow[n\to+\infty]{}e^2 .
\]
Donc $a_n\sim\dfrac{e^2}{n^2}$, terme général positif d'une série de Riemann convergente ($2>1$) : \textbf{la série converge}.

\remarque{La règle de d'Alembert ne conclut pas ici : $\frac{a_{n+1}}{a_n}\to1$ (et non $0$).}

\underline{$\sum\dfrac{\sqrt{5n+1}}{n!}$.} Termes positifs et
\[
\frac{a_{n+1}}{a_n}=\frac{\sqrt{5n+6}}{(n+1)!}\cdot\frac{n!}{\sqrt{5n+1}}=\frac{1}{n+1}\sqrt{\frac{5n+6}{5n+1}}\xrightarrow[n\to+\infty]{}0<1 .
\]
D'après la règle de d'Alembert, \textbf{la série converge}.

\textbf{(b) Rayon de convergence de $\sum\dfrac{3^n(n+7)^3}{7^{n+3}}x^n$.}

Avec $c_n=\dfrac{3^n(n+7)^3}{7^{n+3}}>0$ :
\[
\frac{c_{n+1}}{c_n}=\frac37\Big(\frac{n+8}{n+7}\Big)^3\xrightarrow[n\to+\infty]{}\frac37 .
\]
D'après la règle de d'Alembert, $\boxed{R=\dfrac73}$.

\textbf{3. $f(x,y)=\dfrac{2\sqrt x+3\sqrt y}{3x^2+xy+2y^2}$ sur $(\mathbb{R}_+^*)^2$.}

\textbf{(a) Homogénéité.} Pour $\lambda>0$ :
\[
f(\lambda x,\lambda y)=\frac{\sqrt\lambda\,(2\sqrt x+3\sqrt y)}{\lambda^2(3x^2+xy+2y^2)}=\lambda^{-3/2}f(x,y).
\]
$f$ est positivement homogène de degré $\boxed{\alpha=-\frac32}$.

\textbf{(b) Degré de $\Big(\dfrac{\partial^2f}{\partial x\partial y}\Big)^3$.} En dérivant $f(\lambda x,\lambda y)=\lambda^\alpha f(x,y)$ par rapport à $x$ : $\lambda\frac{\partial f}{\partial x}(\lambda x,\lambda y)=\lambda^\alpha\frac{\partial f}{\partial x}(x,y)$, donc $\frac{\partial f}{\partial x}$ est homogène de degré $\alpha-1$. De même $\frac{\partial^2f}{\partial x\partial y}$ est homogène de degré $\alpha-2=-\frac72$, et son cube est homogène de degré $3\times\big(-\frac72\big)=\boxed{-\frac{21}{2}}$.

\textbf{(c) Calcul de $\dfrac{x^2}{f}\dfrac{\partial^2f}{\partial x^2}+\dfrac{2xy}{f}\dfrac{\partial^2f}{\partial x\partial y}+\dfrac{y^2}{f}\dfrac{\partial^2f}{\partial y^2}$.}

\underline{Formule d'Euler.} Si $f$ est homogène de degré $\alpha$ et de classe $C^1$, alors $xf'_x+yf'_y=\alpha f$ (on dérive $f(\lambda x,\lambda y)=\lambda^\alpha f(x,y)$ par rapport à $\lambda$ puis on fait $\lambda=1$).

\underline{Ordre 2.} On applique Euler à $f'_x$ et $f'_y$, homogènes de degré $\alpha-1$ :
\[
xf''_{xx}+yf''_{xy}=(\alpha-1)f'_x,\qquad xf''_{yx}+yf''_{yy}=(\alpha-1)f'_y .
\]
En multipliant la première par $x$, la seconde par $y$, en additionnant et en utilisant le théorème de Schwarz ($f''_{xy}=f''_{yx}$) :
\[
x^2f''_{xx}+2xyf''_{xy}+y^2f''_{yy}=(\alpha-1)(xf'_x+yf'_y)=\alpha(\alpha-1)f .
\]
Comme $f>0$ :
\[
\frac{x^2}{f}\frac{\partial^2f}{\partial x^2}+\frac{2xy}{f}\frac{\partial^2f}{\partial x\partial y}+\frac{y^2}{f}\frac{\partial^2f}{\partial y^2}=\alpha(\alpha-1)=\Big(-\frac32\Big)\Big(-\frac52\Big)=\boxed{\frac{15}{4}}.
\]

\medskip\textbf{Exercice 2}

\textbf{(1)(a) Loi de $X$.} Les 4 appels sont indépendants et chacun est en retard avec la probabilité $p=0{,}25$ : $X$ compte le nombre de « succès » dans 4 épreuves de Bernoulli indépendantes, donc \textbf{$X\sim\mathcal{B}(4\,;\,0{,}25)$} :
\[
P(X=k)=C_4^k(0{,}25)^k(0{,}75)^{4-k},\qquad k\in\{0,1,2,3,4\}.
\]
\begin{center}
\resizebox{\textwidth}{!}{\begin{tabular}{|c|c|c|c|c|c|}
\hline
$k$ & 0 & 1 & 2 & 3 & 4\\
\hline
$P(X=k)$ & $\frac{81}{256}\approx0{,}3164$ & $\frac{108}{256}\approx0{,}4219$ & $\frac{54}{256}\approx0{,}2109$ & $\frac{12}{256}\approx0{,}0469$ & $\frac{1}{256}\approx0{,}0039$\\
\hline
\end{tabular}}
\end{center}
$E(X)=np=4\times0{,}25=\boxed{1}$ \quad et \quad $V(X)=np(1-p)=4\times0{,}25\times0{,}75=\boxed{0{,}75}$.

\textbf{(b)} $P(X\ge1)=1-P(X=0)=1-\dfrac{81}{256}=\dfrac{175}{256}\approx\boxed{0{,}6836}$.

\textbf{(2)(a) Loi conditionnelle.} Sachant $Y=n$, il y a $n$ appels, chacun en retard avec la probabilité $p=0{,}25$ indépendamment : sachant $Y=n$, $Z$ suit la loi $\mathcal{B}(n,p)$ :
\[
P(Z=k\mid Y=n)=C_n^kp^k(1-p)^{n-k}\quad\text{si }0\le k\le n,\qquad P(Z=k\mid Y=n)=0\quad\text{si }k>n .
\]

\textbf{(b) Loi du couple.} Avec $P(Y=n)=e^{-m}\dfrac{m^n}{n!}$, pour $0\le k\le n$ :
\begin{align*}
P(Z=k,Y=n)&=P(Z=k\mid Y=n)\,P(Y=n)=\frac{n!}{k!(n-k)!}p^k(1-p)^{n-k}e^{-m}\frac{m^n}{n!}\\
&=e^{-m}\frac{(mp)^k\,\big(m(1-p)\big)^{n-k}}{k!\,(n-k)!}.
\end{align*}

\textbf{(c) Loi de $Z$.} Par la formule des probabilités totales (système complet $\{Y=n\}_{n\in\mathbb{N}}$) :
\begin{align*}
P(Z=k)&=\sum_{n=k}^{+\infty}P(Z=k,Y=n)=e^{-m}\frac{(mp)^k}{k!}\sum_{n=k}^{+\infty}\frac{\big(m(1-p)\big)^{n-k}}{(n-k)!}\\
&\underset{j=n-k}{=}e^{-m}\frac{(mp)^k}{k!}\sum_{j=0}^{+\infty}\frac{\big(m(1-p)\big)^{j}}{j!}.
\end{align*}
La dernière somme est le développement en série de l'exponentielle : $\sum_{j\ge0}\frac{t^j}{j!}=e^t$ pour tout réel $t$. Donc
\[
P(Z=k)=e^{-m}\frac{(mp)^k}{k!}e^{m(1-p)}=e^{-mp}\frac{(mp)^k}{k!},\qquad k\in\mathbb{N}.
\]
\textbf{$Z$ suit la loi de Poisson de paramètre $mp=0{,}25\,m$.}

\textbf{(3) Loi de $U$.} $U$ est le rang du premier appel traité en retard dans une suite d'appels indépendants, chacun en retard avec la probabilité $p=0{,}25$ : \textbf{$U$ suit la loi géométrique de paramètre $p=0{,}25$} sur $\mathbb{N}^*$ :
\[
P(U=k)=(1-p)^{k-1}p=0{,}25\times0{,}75^{\,k-1},\qquad k\ge1,
\]
(les $k-1$ premiers appels sont à l'heure, le $k$-ième en retard), et
\[
E(U)=\frac1p=\boxed{4}.
\]
En moyenne, le premier retard survient au 4\textsuperscript{e} appel.

\medskip\textbf{Exercice 3}

$f(x)=\displaystyle\int_x^{2x}\frac{du}{u+\sin u}$, $x\neq0$.

\textbf{1.a) Existence.} Pour $u\neq0$, $|\sin u|<|u|$, donc $u+\sin u$ a le signe de $u$ et ne s'annule qu'en $u=0$. Pour $x\neq0$, le segment d'extrémités $x$ et $2x$ ne contient pas $0$ ; la fonction $u\mapsto\frac{1}{u+\sin u}$ y est continue, donc $f(x)$ existe.

\textbf{b) Parité.} Pour $x\neq0$, posons $u=-v$ ($du=-dv$) dans $f(-x)$ :
\[
f(-x)=\int_{-x}^{-2x}\frac{du}{u+\sin u}=\int_{x}^{2x}\frac{-dv}{-v+\sin(-v)}=\int_x^{2x}\frac{-dv}{-(v+\sin v)}=\int_x^{2x}\frac{dv}{v+\sin v}=f(x).
\]
\textbf{$f$ est paire.}

\textbf{c) Signe.} Pour $x>0$ et $u\in[x,2x]$, $u+\sin u>0$, donc $\frac1{u+\sin u}>0$. C'est une fonction continue et strictement positive intégrée sur $[x,2x]$ avec $x<2x$ : $f(x)>0$.

\textbf{2. Dérivée et variations ($x>0$).} Soit $F$ une primitive de $u\mapsto\frac1{u+\sin u}$ sur $]0,+\infty[$ ; $f(x)=F(2x)-F(x)$, donc
\[
f'(x)=\frac{2}{2x+\sin2x}-\frac{1}{x+\sin x}=\frac{2(x+\sin x)-(2x+\sin2x)}{(x+\sin x)(2x+\sin2x)}=\frac{2\sin x-2\sin x\cos x}{(x+\sin x)(2x+\sin2x)},
\]
\[
\boxed{f'(x)=\frac{2\sin x\,(1-\cos x)}{(x+\sin x)(2x+\sin2x)}}.
\]
Le dénominateur est strictement positif et $1-\cos x\ge0$ : \textbf{$f'(x)$ a le signe de $\sin x$} (et s'annule aux points $k\pi$).
\begin{itemize}
\item Sur $[2k\pi,(2k+1)\pi]$ ($k\in\mathbb{N}$), $\sin x\ge0$ : $f$ est croissante.
\item Sur $[(2k+1)\pi,(2k+2)\pi]$, $\sin x\le0$ : $f$ est décroissante.
\end{itemize}
$f$ n'est donc \emph{pas} monotone sur $]0,+\infty[$ : elle oscille, avec des maxima locaux en $(2k+1)\pi$ et des minima locaux en $2k\pi$ ($k\ge1$).

\textbf{3.(a) Encadrements.} Pour $u>1$ : $-1\le\sin u\le1$ donne $0<u-1\le u+\sin u\le u+1$, donc
\[
\frac1{u+1}\le\frac1{u+\sin u}\le\frac1{u-1}.
\]
Pour $x>1$, tout $u\in[x,2x]$ vérifie $u>1$ ; en intégrant (bornes dans l'ordre croissant) :
\[
\int_x^{2x}\frac{du}{u+1}\le f(x)\le\int_x^{2x}\frac{du}{u-1},\qquad\boxed{\ln\frac{2x+1}{x+1}\le f(x)\le\ln\frac{2x-1}{x-1}}.
\]

\textbf{(b) Limite en $+\infty$.} $\dfrac{2x+1}{x+1}\to2$ et $\dfrac{2x-1}{x-1}\to2$, et $\ln$ est continue en $2$ : les deux bornes tendent vers $\ln2$. Par le théorème des gendarmes,
\[
\boxed{\lim_{x\to+\infty}f(x)=\ln2}.
\]

\medskip\textbf{Exercice 4}

\emph{Rappel (sommes de Riemann).} Si $g$ est continue sur $[0,1]$, alors $\displaystyle\frac1n\sum_{k=0}^{n-1}g\Big(\frac kn\Big)\to\int_0^1g$ et $\displaystyle\frac1n\sum_{k=1}^{n}g\Big(\frac kn\Big)\to\int_0^1g$.

\textbf{1.} En factorisant $n^2$ au dénominateur :
\[
u_n=n\sum_{k=0}^{n-1}\frac{1}{n^2\big(1+\frac{k^2}{n^2}\big)}=\frac1n\sum_{k=0}^{n-1}g\Big(\frac kn\Big),\qquad g(t)=\frac1{1+t^2}.
\]
$g$ est continue sur $[0,1]$, donc
\[
\lim_{n\to+\infty}u_n=\int_0^1\frac{dt}{1+t^2}=\Big[\arctan t\Big]_0^1=\boxed{\frac\pi4}.
\]

\textbf{2.} $v_n>0$ et
\[
\ln v_n=\frac1n\sum_{k=1}^n\ln\Big(1+\frac{k^2}{n^2}\Big)=\frac1n\sum_{k=1}^nh\Big(\frac kn\Big),\qquad h(t)=\ln(1+t^2),
\]
continue sur $[0,1]$. Donc $\ln v_n\to\displaystyle\int_0^1\ln(1+t^2)\,dt$. Par parties ($u=\ln(1+t^2)$, $v'=1$) :
\[
\int_0^1\ln(1+t^2)\,dt=\Big[t\ln(1+t^2)\Big]_0^1-\int_0^1\frac{2t^2}{1+t^2}dt=\ln2-2\int_0^1\Big(1-\frac1{1+t^2}\Big)dt=\ln2-2+\frac\pi2 .
\]
Par continuité de l'exponentielle :
\[
\boxed{\lim_{n\to+\infty}v_n=e^{\ln2-2+\pi/2}=2\,e^{\frac\pi2-2}\approx1{,}302}.
\]
\remarque{On ne peut pas remplacer $\ln\big(1+\frac{k^2}{n^2}\big)$ par $\frac{k^2}{n^2}$ : le quotient $\frac kn$ n'est pas petit (il varie entre $0$ et $1$), et un équivalent ne s'additionne pas de cette façon dans une somme dont le nombre de termes varie. Ce raisonnement conduirait au résultat faux $e^{1/3}$.}
