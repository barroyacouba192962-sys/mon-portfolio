\textbf{Exercice 1}

$T(e_1)=T(e_3)=e_3$ et $T(e_2)=-e_1+e_2+e_3$.

\textbf{1.a) Noyau de $T$.}

Soit $U=xe_1+ye_2+ze_3\in\mathbb{R}^3$. Par linéarité :
\[
T(U)=xT(e_1)+yT(e_2)+zT(e_3)=xe_3+y(-e_1+e_2+e_3)+ze_3=-y\,e_1+y\,e_2+(x+y+z)\,e_3 .
\]
Comme $(e_1,e_2,e_3)$ est une base :
\[
U\in\ker T\iff\begin{cases}-y=0\\y=0\\x+y+z=0\end{cases}\iff\begin{cases}y=0\\z=-x\end{cases}\iff U=x(e_1-e_3),\ x\in\mathbb{R}.
\]
\[
\boxed{\ker T=\operatorname{Vect}(e_1-e_3)}
\]
est une droite vectorielle. D'après le théorème du rang, $\operatorname{rg}T=3-1=2$.

\textbf{1.b) Matrice de $T$ dans la base $\mathcal{B}=(e_1,e_2,e_3)$.}

La $j$-ème colonne de $A$ contient les coordonnées de $T(e_j)$ dans $\mathcal{B}$ : $T(e_1)=(0,0,1)$, $T(e_2)=(-1,1,1)$, $T(e_3)=(0,0,1)$. D'où
\[
A=\begin{pmatrix}
0 & -1 & 0 \\
0 & 1 & 0 \\
1 & 1 & 1
\end{pmatrix}.
\]

\textbf{2.a) Expression des $e_i$ en fonction des $f_i$ ; $(f_1,f_2,f_3)$ est une base.}
\[
\begin{cases}
f_1=e_1-e_3\\
f_2=e_1-e_2\\
f_3=-e_1+e_2+e_3
\end{cases}
\]
En additionnant les deux dernières équations : $f_2+f_3=e_3$. Puis $e_1=f_1+e_3=f_1+f_2+f_3$, et $e_2=e_1-f_2=f_1+f_3$ :
\[
\boxed{e_1=f_1+f_2+f_3,\qquad e_2=f_1+f_3,\qquad e_3=f_2+f_3}.
\]
Ainsi $e_1,e_2,e_3\in\operatorname{Vect}(f_1,f_2,f_3)$, donc $\mathbb{R}^3=\operatorname{Vect}(e_1,e_2,e_3)\subset\operatorname{Vect}(f_1,f_2,f_3)$ : la famille $(f_1,f_2,f_3)$ est génératrice de $\mathbb{R}^3$. Comme elle a $3=\dim\mathbb{R}^3$ éléments, c'est une \textbf{base} de $\mathbb{R}^3$, que l'on note $\mathcal{B}'$.

\textbf{2.b) Matrice de $T$ dans la base $\mathcal{B}'=(f_1,f_2,f_3)$.}
\begin{itemize}
\item $T(f_1)=T(e_1)-T(e_3)=e_3-e_3=0$ (on savait déjà que $f_1=e_1-e_3\in\ker T$) ;
\item $T(f_2)=T(e_1)-T(e_2)=e_3-(-e_1+e_2+e_3)=e_1-e_2=f_2$ ;
\item $T(f_3)=-T(e_1)+T(e_2)+T(e_3)=-e_3+(-e_1+e_2+e_3)+e_3=-e_1+e_2+e_3=f_3$.
\end{itemize}
Donc
\[
B=\begin{pmatrix}
0 & 0 & 0 \\
0 & 1 & 0 \\
0 & 0 & 1
\end{pmatrix}.
\]
($T$ est la projection sur $\operatorname{Vect}(f_2,f_3)$ parallèlement à $\operatorname{Vect}(f_1)$, car $B^2=B$.)

\textbf{3. La matrice $P$.}

Les colonnes de $P=\begin{pmatrix}1&1&-1\\0&-1&1\\-1&0&1\end{pmatrix}$ sont les coordonnées de $f_1,f_2,f_3$ dans $\mathcal{B}$ : $P$ est la \textbf{matrice de passage} de $\mathcal{B}$ à $\mathcal{B}'$.

\underline{Inversibilité et calcul de $P^{-1}$.} On résout $PX=Y$ pour $Y=(y_1,y_2,y_3)$ quelconque :
\[
\begin{cases}
x_1+x_2-x_3=y_1 & (L_1)\\
-x_2+x_3=y_2 & (L_2)\\
-x_1+x_3=y_3 & (L_3)
\end{cases}
\]
$(L_1)+(L_2)$ donne $x_1=y_1+y_2$ ; puis $(L_3)$ donne $x_3=y_3+x_1=y_1+y_2+y_3$ ; enfin $(L_2)$ donne $x_2=x_3-y_2=y_1+y_3$. Le système a une unique solution pour tout $Y$ : $P$ est inversible, et
\[
\boxed{P^{-1}=\begin{pmatrix}
1 & 1 & 0 \\
1 & 0 & 1 \\
1 & 1 & 1
\end{pmatrix}}.
\]
On peut aussi lire $P^{-1}$ directement sur la question 2.a : ses colonnes sont les coordonnées de $e_1,e_2,e_3$ dans $\mathcal{B}'$. (On vérifie $\det P=1\cdot(-1-0)-1\cdot(0-1)+(-1)\cdot(0-1)=1\neq0$.)

\underline{Relation entre $A$, $B$, $P$, $P^{-1}$.} Par la formule de changement de base :
\[
\boxed{B=P^{-1}AP\qquad\text{ou, de façon équivalente,}\qquad A=PBP^{-1}}.
\]
\emph{Justification.} Soit $u\in\mathbb{R}^3$, de coordonnées $X$ dans $\mathcal{B}$ et $X'$ dans $\mathcal{B}'$ ; on a $X=PX'$. Les coordonnées de $T(u)$ sont $AX$ dans $\mathcal{B}$ et $BX'$ dans $\mathcal{B}'$, donc $AX=P(BX')$, soit $APX'=PBX'$ pour tout $X'$. Ainsi $AP=PB$.

\medskip\textbf{Exercice 2}

$T$ : $M(z)\mapsto M'(z')$ avec $z'=(1+i)z-i$.

\textbf{1. Nature de $T$, point invariant, angle $(\overrightarrow{\Omega M},\overrightarrow{MM'})$.}

\underline{Point invariant.} $z=(1+i)z-i\iff iz=i\iff z=1$. Le point invariant est $\boxed{\Omega(1)}$.

\underline{Forme réduite.} En soustrayant $1=(1+i)\cdot1-i$ :
\[
z'-1=(1+i)(z-1),\qquad 1+i=\sqrt2\,e^{i\pi/4}.
\]
$T$ est donc la similitude directe de centre $\Omega$, de rapport $\sqrt2$ et d'angle $\frac\pi4$. C'est la composée (dans un ordre quelconque, elles commutent) :
\begin{itemize}
\item de la rotation $r$ de centre $\Omega$ et d'angle $\frac\pi4$ : $z\mapsto 1+e^{i\pi/4}(z-1)$ ;
\item de l'homothétie $h$ de centre $\Omega$ et de rapport $\sqrt2>0$ : $z\mapsto1+\sqrt2(z-1)$.
\end{itemize}

\underline{Angle.} Pour $M\neq\Omega$ :
\[
z'-z=(1+i)(z-1)-(z-1)=i(z-1),\qquad\text{donc}\qquad\frac{z'-z}{z-1}=i=e^{i\pi/2}.
\]
Ainsi $\boxed{(\overrightarrow{\Omega M},\overrightarrow{MM'})=\dfrac\pi2\ [2\pi]}$ et $MM'=\Omega M$ : le triangle $\Omega MM'$ est \textbf{rectangle isocèle en $M$}.

\textbf{2.a) Construction de $M'$.} D'après la question 1, $M'$ est le point tel que $\overrightarrow{MM'}$ s'obtienne en tournant $\overrightarrow{\Omega M}$ de $+\frac\pi2$ : on trace la perpendiculaire à $(\Omega M)$ en $M$, et on y place $M'$ tel que $MM'=M\Omega$, dans le sens direct. (Exemple : $M(2+i)$ donne $M'(1+2i)$.)

\textbf{2.b) Image de la droite $(D)$ : $y=x$.}

Un point de $(D)$ a pour affixe $z=t(1+i)$, $t\in\mathbb{R}$. Alors
\[
z'=(1+i)\,t(1+i)-i=t(1+i)^2-i=2it-i=i(2t-1).
\]
Quand $t$ décrit $\mathbb{R}$, $2t-1$ décrit $\mathbb{R}$ : $z'$ décrit l'ensemble des imaginaires purs. Donc
\[
\boxed{(D')\ :\ x=0}\quad\text{(l'axe des ordonnées).}
\]
(Cohérent avec l'angle $\frac\pi4$ : la droite de pente $1$, tournée de $\frac\pi4$, devient verticale.)

\textbf{3.a) Le point $B$.}

On cherche $z_0\neq1$ tel que $z_0z_0'=1$ avec $z_0'=(1+i)z_0-i$ :
\[
z_0\big((1+i)z_0-i\big)=1\iff(1+i)z_0^2-iz_0-1=0 .
\]
Discriminant : $\Delta=(-i)^2-4(1+i)(-1)=-1+4+4i=3+4i=(2+i)^2$ (car $(2+i)^2=4+4i-1$). Les racines sont
\[
z=\frac{i\pm(2+i)}{2(1+i)}:\qquad z=\frac{2+2i}{2(1+i)}=1\quad\text{ou}\quad z=\frac{-2}{2(1+i)}=\frac{-1}{1+i}=\frac{-1+i}{2}.
\]
La racine $z=1$ correspond à $\Omega$, qui est exclu. Il existe donc un unique point $B$ distinct de $\Omega$ :
\[
\boxed{z_0=\frac{-1+i}{2}},\qquad z_0'=\frac1{z_0}=\frac{2}{-1+i}=\frac{2(-1-i)}{2}=\boxed{-1-i}.
\]
(Vérification : $(1+i)\frac{-1+i}{2}-i=\frac{-2}{2}-i=-1-i$.)

\textbf{3.b) Cocyclicité de $\Omega'$, $\Omega$, $B$, $B'$ et centre $G$.}

$\Omega'$, symétrique de $\Omega$ par rapport à $O$, a pour affixe $-1$. Les points sont $\Omega'(-1)$, $\Omega(1)$, $B\big(-\frac12+\frac12i\big)$, $B'(-1-i)$.

Cherchons un point $G$ équidistant des quatre points. Comme $G$ est équidistant de $\Omega(1,0)$ et $\Omega'(-1,0)$, il est sur la médiatrice de $[\Omega\Omega']$ : l'axe des ordonnées, $G(0,c)$. L'égalité $G\Omega^2=GB'^2$ s'écrit
\[
1+c^2=1+(c+1)^2\iff 0=2c+1\iff c=-\frac12 .
\]
Avec $G\big(0,-\frac12\big)$ : $G\Omega^2=G\Omega'^2=GB'^2=1+\frac14=\frac54$ et
\[
GB^2=\Big(-\frac12\Big)^2+\Big(\frac12+\frac12\Big)^2=\frac14+1=\frac54 .
\]
Les quatre points sont à la même distance $\frac{\sqrt5}{2}$ de $G$ : ils sont \textbf{cocycliques}, sur le cercle $(\Gamma)$ de centre $G$ d'affixe
\[
\boxed{z_G=-\frac i2}\qquad\text{et de rayon }\frac{\sqrt5}{2}.
\]

\medskip\textbf{Exercice 3}

\textbf{1. $I=\displaystyle\iint_D\frac{dx\,dy}{(1+x^2+y^2)^2}$, \quad $D=\{(x,y)\ /\ |x|\le x^2+y^2\le1\}$.}

\underline{Description de $D$ en polaires} ($x=r\cos\theta$, $y=r\sin\theta$, $r\ge0$, $dx\,dy=r\,dr\,d\theta$) :
\[
|x|\le x^2+y^2\iff r|\cos\theta|\le r^2\iff r\ge|\cos\theta|\ \ (\text{pour } r>0),
\]
\[
x^2+y^2\le1\iff r\le1 .
\]
Donc $D=\{\,|\cos\theta|\le r\le1,\ \theta\in[0,2\pi]\,\}$ : c'est le disque unité privé des deux disques de diamètres $[O,(1,0)]$ et $[O,(-1,0)]$ (la condition $|x|\le x^2+y^2$ ne doit pas être oubliée).

\underline{Calcul.} Une primitive de $\dfrac{r}{(1+r^2)^2}$ est $-\dfrac{1}{2(1+r^2)}$. Par le théorème de Fubini :
\[
I=\int_0^{2\pi}\Big[-\frac{1}{2(1+r^2)}\Big]_{|\cos\theta|}^{1}d\theta=\int_0^{2\pi}\Big(\frac{1}{2(1+\cos^2\theta)}-\frac14\Big)d\theta=\frac12K-\frac\pi2,
\]
où $K=\displaystyle\int_0^{2\pi}\frac{d\theta}{1+\cos^2\theta}$. Par périodicité et symétries de $\cos^2$, $K=4\displaystyle\int_0^{\pi/2}\frac{d\theta}{1+\cos^2\theta}$. Sur $[0,\frac\pi2[$, on pose $t=\tan\theta$ : $d\theta=\dfrac{dt}{1+t^2}$ et $\cos^2\theta=\dfrac1{1+t^2}$, donc $1+\cos^2\theta=\dfrac{2+t^2}{1+t^2}$ et
\[
\int_0^{\pi/2}\frac{d\theta}{1+\cos^2\theta}=\int_0^{+\infty}\frac{dt}{2+t^2}=\Big[\frac1{\sqrt2}\arctan\frac{t}{\sqrt2}\Big]_0^{+\infty}=\frac{\pi}{2\sqrt2}.
\]
Ainsi $K=\dfrac{4\pi}{2\sqrt2}=\pi\sqrt2$ et
\[
\boxed{I=\frac{\pi\sqrt2}{2}-\frac\pi2=\frac{\pi(\sqrt2-1)}{2}\approx0{,}651}.
\]

\textbf{2. $J=\displaystyle\iint_D(x+y)^2\,dx\,dy$, \quad $D=\{x^2+y^2-x\le0,\ x^2+y^2-y\ge0,\ y\ge0\}$.}

\underline{Description de $D$.}
\begin{itemize}
\item $x^2+y^2\le x\iff(x-\frac12)^2+y^2\le\frac14$ : disque de centre $(\frac12,0)$ de rayon $\frac12$ ; en polaires $r\le\cos\theta$.
\item $x^2+y^2\ge y\iff x^2+(y-\frac12)^2\ge\frac14$ : extérieur du disque de centre $(0,\frac12)$ de rayon $\frac12$ ; en polaires $r\ge\sin\theta$.
\item $y\ge0$ : $\theta\in[0,\pi]$ ; avec $r\le\cos\theta$ il faut de plus $\cos\theta\ge0$, donc $\theta\in[0,\frac\pi2]$.
\end{itemize}
La condition $\sin\theta\le r\le\cos\theta$ n'est possible que si $\sin\theta\le\cos\theta$, c'est-à-dire $\theta\in[0,\frac\pi4]$. Donc
\[
D=\Big\{\,0\le\theta\le\frac\pi4,\ \sin\theta\le r\le\cos\theta\,\Big\}.
\]
(Les deux cercles se coupent en $O$ et en $(\frac12,\frac12)$, point d'angle polaire $\frac\pi4$.)

\underline{Calcul.} $(x+y)^2=r^2(\cos\theta+\sin\theta)^2=r^2(1+\sin2\theta)$, d'où
\[
J=\int_0^{\pi/4}(1+\sin2\theta)\int_{\sin\theta}^{\cos\theta}r^3\,dr\,d\theta=\int_0^{\pi/4}(1+\sin2\theta)\,\frac{\cos^4\theta-\sin^4\theta}{4}\,d\theta .
\]
Or $\cos^4\theta-\sin^4\theta=(\cos^2\theta-\sin^2\theta)(\cos^2\theta+\sin^2\theta)=\cos2\theta$. Donc
\[
J=\frac14\int_0^{\pi/4}\big(\cos2\theta+\sin2\theta\cos2\theta\big)\,d\theta=\frac14\Big[\frac{\sin2\theta}{2}+\frac{\sin^22\theta}{4}\Big]_0^{\pi/4}=\frac14\Big(\frac12+\frac14\Big),
\]
\[
\boxed{J=\frac{3}{16}}.
\]

\textbf{3. $S=\displaystyle\iiint_D\frac{z^3}{(x+y+z)(y+z)}\,dx\,dy\,dz$, \quad $D=\{x,y,z\ge0,\ x+y+z\le1\}$.}

\underline{Changement de variables.} $u=x+y+z$, $v=\dfrac{z}{y+z}$, $w=\dfrac{y+z}{x+y+z}$. On exprime $(x,y,z)$ en fonction de $(u,v,w)$ :
\[
y+z=uw,\qquad z=v(y+z)=uvw,\qquad y=uw-uvw=uw(1-v),\qquad x=u-uw=u(1-w).
\]
\underline{Nouveau domaine.} $x,y,z\ge0$ et $x+y+z\le1$ équivaut (à un ensemble de mesure nulle près) à $(u,v,w)\in\Delta=[0,1]^3$.

\underline{Jacobien.}
\[
\frac{D(x,y,z)}{D(u,v,w)}=\begin{vmatrix}
1-w & 0 & -u\\
w(1-v) & -uw & u(1-v)\\
vw & uw & uv
\end{vmatrix}.
\]
En ajoutant la ligne 2 à la ligne 3, puis la ligne 1 à la nouvelle ligne 3, la ligne 3 devient $(1,0,0)$. En développant suivant cette ligne :
\[
\frac{D(x,y,z)}{D(u,v,w)}=\begin{vmatrix}0&-u\\-uw&u(1-v)\end{vmatrix}=-u^2w,\qquad\text{donc}\qquad dx\,dy\,dz=u^2w\,du\,dv\,dw .
\]
\underline{Intégrande.} $\dfrac{z^3}{(x+y+z)(y+z)}=\dfrac{u^3v^3w^3}{u\cdot uw}=uv^3w^2$.

\underline{Calcul.}
\[
S=\iiint_{[0,1]^3}uv^3w^2\cdot u^2w\,du\,dv\,dw=\int_0^1u^3\,du\int_0^1v^3\,dv\int_0^1w^3\,dw=\Big(\frac14\Big)^3,
\]
\[
\boxed{S=\frac1{64}}.
\]

\textbf{4. Aire de l'ellipse $E=\Big\{\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}\le1\Big\}$ ($a,b>0$).}

On pose $x=a\rho\cos\theta$, $y=b\rho\sin\theta$ (coordonnées polaires « elliptiques »), avec $\rho\in[0,1]$, $\theta\in[0,2\pi]$. Le jacobien vaut
\[
\frac{D(x,y)}{D(\rho,\theta)}=\begin{vmatrix}a\cos\theta&-a\rho\sin\theta\\b\sin\theta&b\rho\cos\theta\end{vmatrix}=ab\rho .
\]
Donc
\[
\mathcal{A}(E)=\iint_Edx\,dy=\int_0^{2\pi}\int_0^1ab\rho\,d\rho\,d\theta=ab\times2\pi\times\frac12,\qquad\boxed{\mathcal{A}(E)=\pi ab}.
\]
(Pour $a=b=R$ on retrouve l'aire $\pi R^2$ du disque.)

\medskip\textbf{Exercice 4}

$f(n)=\displaystyle\int_0^1x^ne^{-x}\,dx$, $n\in\mathbb{N}$.

\textbf{1. Positivité, décroissance, relation de récurrence.}

\underline{Positivité.} Sur $[0,1]$, $x^ne^{-x}\ge0$ ; la fonction est continue, positive et non identiquement nulle, donc $f(n)>0$.

\underline{Décroissance.} Pour $x\in[0,1]$, $x^{n+1}\le x^n$, donc $x^{n+1}e^{-x}\le x^ne^{-x}$ et, par croissance de l'intégrale, $f(n+1)\le f(n)$.

\underline{Récurrence.} Pour $n\ge1$, on intègre par parties avec $u(x)=x^n$, $u'(x)=nx^{n-1}$, $v'(x)=e^{-x}$, $v(x)=-e^{-x}$ :
\[
f(n)=\Big[-x^ne^{-x}\Big]_0^1+n\int_0^1x^{n-1}e^{-x}\,dx,\qquad\boxed{f(n)=nf(n-1)-\frac1e}.
\]

\textbf{2. Formule explicite.} Montrons par récurrence $\mathcal{P}(n)$ : $f(n)=\dfrac{n!}{e}\Big(e-\displaystyle\sum_{k=0}^n\frac1{k!}\Big)$.

\emph{Initialisation.} $f(0)=\displaystyle\int_0^1e^{-x}dx=1-\frac1e$, et $\dfrac{0!}{e}(e-1)=1-\dfrac1e$.

\emph{Hérédité.} Si $\mathcal{P}(n-1)$ est vraie ($n\ge1$), alors
\[
f(n)=n\cdot\frac{(n-1)!}{e}\Big(e-\sum_{k=0}^{n-1}\frac1{k!}\Big)-\frac1e=\frac{n!}{e}\Big(e-\sum_{k=0}^{n-1}\frac1{k!}-\frac1{n!}\Big)=\frac{n!}{e}\Big(e-\sum_{k=0}^{n}\frac1{k!}\Big).
\]
La propriété est vraie pour tout $n\in\mathbb{N}$.

\textbf{3. Encadrement.} Pour $x\in[0,1]$, $e^{-1}\le e^{-x}\le1$, donc $\dfrac{x^n}{e}\le x^ne^{-x}\le x^n$. En intégrant sur $[0,1]$, avec $\displaystyle\int_0^1x^n\,dx=\frac1{n+1}$ :
\[
\boxed{\frac{1}{e(n+1)}\le f(n)\le\frac1{n+1}}.
\]
En particulier $f(n)\to0$.

\textbf{4. Nature des séries.}
\begin{itemize}
\item $\sum f(n)$ : $f(n)\ge\dfrac{1}{e(n+1)}\ge0$ et $\sum\dfrac{1}{n+1}$ diverge (série harmonique). Par comparaison de séries à termes positifs, \textbf{$\sum f(n)$ diverge}.
\item $\sum\dfrac{f(n)}{n}$ : $0\le\dfrac{f(n)}{n}\le\dfrac1{n(n+1)}\le\dfrac1{n^2}$, et $\sum\dfrac1{n^2}$ converge (série de Riemann, $2>1$). \textbf{$\sum\dfrac{f(n)}{n}$ converge}.
\item $\sum(-1)^nf(n)$ : la suite $(f(n))$ est positive, décroissante et tend vers $0$. D'après le critère spécial des séries alternées, \textbf{$\sum(-1)^nf(n)$ converge}.
\end{itemize}

\textbf{5. Rayon de convergence $R$ de $\sum f(n)x^n$.}

\begin{itemize}
\item En $x=-1$, la série $\sum f(n)(-1)^n$ converge : donc $R\ge1$ (une série entière diverge grossièrement en tout point de module $>R$, donc tout point de convergence vérifie $|x|\le R$).
\item En $x=1$, la série $\sum f(n)$ diverge : donc $R\le1$ (une série entière converge absolument en tout point de module $<R$).
\end{itemize}
Donc $\boxed{R=1}$. (Autre méthode : l'encadrement de la question 3 et le fait que $\sum\frac{x^n}{n+1}$ a pour rayon $1$.)
