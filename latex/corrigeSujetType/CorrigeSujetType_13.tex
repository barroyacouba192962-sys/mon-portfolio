\textbf{Exercice 1}

\textbf{Partie 1}

\remarque{Pour que la condition (ii) $t_0<t_1<\dots<t_n$ soit compatible avec $t_0=0$ et $t_n=A$, il faut $A>0$ : on suppose donc $A$ strictement positif (l'énoncé original disait seulement « non nul »).}

\underline{Unicité.} Soit $(t_0,\dots,t_n)$ une suite vérifiant (i), (ii), (iii). D'après (iii), $(t_k)$ est une suite arithmétique de raison $\frac An$ et de premier terme $t_0=0$ ; donc, par récurrence immédiate, $t_k=t_0+k\frac An=\dfrac{kA}{n}$ pour tout $k\in\{0,\dots,n\}$. La suite est donc nécessairement celle-ci.

\underline{Existence.} Réciproquement, posons $t_k=\dfrac{kA}{n}$ ($0\le k\le n$). Alors $t_0=0$, $t_n=A$ (i) ; $t_{k+1}-t_k=\frac An>0$ (iii), donc la suite est strictement croissante (ii).

Il existe donc une unique suite convenable : $\boxed{t_k=\dfrac{kA}{n}}$, $k=0,\dots,n$.

\textbf{Partie 2}

\textbf{1) Existence et unicité des $x_k$.}

Posons $F(x)=\displaystyle\int_0^xf(t)\,dt$ pour $x\in[0,1]$ et $A=F(1)=\displaystyle\int_0^1f(t)\,dt$. Comme $f$ est continue, $F$ est dérivable de dérivée $F'=f>0$ : $F$ est continue et strictement croissante sur $[0,1]$, donc réalise une bijection de $[0,1]$ sur $[F(0),F(1)]=[0,A]$, et $A>0$.

Pour $x_0,\dots,x_n\in[0,1]$, posons $t_k=F(x_k)$. Par la relation de Chasles, $\displaystyle\int_{x_k}^{x_{k+1}}f(t)dt=F(x_{k+1})-F(x_k)=t_{k+1}-t_k$. Les conditions de la partie 2 sur les $x_k$ équivalent donc aux conditions de la partie 1 sur les $t_k$ :
\begin{itemize}
\item (i) $x_0=0,\ x_n=1\iff t_0=0,\ t_n=A$ (car $F$ est bijective) ;
\item (ii) $x_0<\dots<x_n\iff t_0<\dots<t_n$ (car $F$ est strictement croissante) ;
\item (iii) $\displaystyle\int_{x_k}^{x_{k+1}}f=\frac1n\int_0^1f\iff t_{k+1}-t_k=\frac An$.
\end{itemize}
D'après la partie 1, il existe une unique suite $(t_k)$ convenable, $t_k=\frac{kA}{n}$, donc une unique suite $(x_k)$ :
\[
\boxed{x_k=F^{-1}\Big(\frac{kA}{n}\Big),\quad k=0,\dots,n}.
\]

\textbf{2) Limite de $U(n)=\dfrac1n\displaystyle\sum_{k=0}^nf(x_k)$.}

Posons $\varphi=f\circ F^{-1}$, continue sur $[0,A]$ (composée de fonctions continues ; $F^{-1}$ est continue comme réciproque d'une bijection continue strictement monotone). Alors
\[
U(n)=\frac1n\sum_{k=0}^n\varphi\Big(\frac{kA}{n}\Big)=\frac1A\cdot\underbrace{\frac An\sum_{k=0}^{n-1}\varphi\Big(\frac{kA}{n}\Big)}_{\text{somme de Riemann}}+\frac{\varphi(A)}{n}.
\]
La somme de Riemann de $\varphi$ sur $[0,A]$, de pas $\frac An$, tend vers $\displaystyle\int_0^A\varphi(t)\,dt$, et $\frac{\varphi(A)}n\to0$. Donc
\[
L=\lim_{n\to+\infty}U(n)=\frac1A\int_0^Af\big(F^{-1}(t)\big)\,dt .
\]
Changement de variable $t=F(u)$, $u\in[0,1]$, $dt=F'(u)\,du=f(u)\,du$ :
\[
\int_0^Af\big(F^{-1}(t)\big)dt=\int_0^1f(u)\,f(u)\,du=\int_0^1f^2(u)\,du,\qquad\boxed{L=\frac{\displaystyle\int_0^1f^2(u)\,du}{\displaystyle\int_0^1f(u)\,du}}.
\]

\textbf{3) Cas $f(x)=e^x$.} $F(x)=e^x-1$, $A=e-1$ ; $x_k=F^{-1}\big(\frac{k(e-1)}{n}\big)=\ln\Big(1+\dfrac{k(e-1)}{n}\Big)$.
\[
\int_0^1e^{2u}du=\frac{e^2-1}{2},\qquad L=\frac{(e^2-1)/2}{e-1}=\boxed{\frac{e+1}{2}\approx1{,}859}.
\]

\medskip\textbf{Exercice 2}

\textbf{1) Étude de $g(x)=\dfrac1{\ln x}$ sur $]0,1[\cup]1,+\infty[$.}

\underline{Dérivées.} $g$ est de classe $C^\infty$ sur son domaine et
\[
g'(x)=-\frac{1/x}{\ln^2x}=-\frac{1}{x\ln^2x}<0,\qquad g''(x)=\frac{\ln^2x+2\ln x}{x^2\ln^4x}=\frac{\ln x+2}{x^2\ln^3x}.
\]
\underline{Sens de variation.} $g'<0$ : $g$ est strictement décroissante sur $]0,1[$ et sur $]1,+\infty[$ (mais pas sur la réunion !).

\underline{Concavité.} Signe de $g''$ = signe de $\frac{\ln x+2}{\ln^3x}$ :
\begin{itemize}
\item sur $]0,e^{-2}[$ : $\ln x+2<0$ et $\ln^3x<0$, donc $g''>0$ : $g$ est convexe ;
\item sur $]e^{-2},1[$ : $\ln x+2>0$ et $\ln^3x<0$, donc $g''<0$ : $g$ est concave ;
\item sur $]1,+\infty[$ : $g''>0$ : $g$ est convexe.
\end{itemize}
La courbe admet un point d'inflexion en $x=e^{-2}\approx0{,}135$, d'ordonnée $g(e^{-2})=-\frac12$.

\underline{Limites.}
\begin{itemize}
\item $x\to0^+$ : $\ln x\to-\infty$, donc $g(x)\to0^-$. De plus $g'(x)\to-\infty$ (car $x\ln^2x\to0^+$) : la courbe arrive en l'origine avec une tangente verticale.
\item $x\to1^-$ : $\ln x\to0^-$, $g(x)\to-\infty$ ; \quad $x\to1^+$ : $\ln x\to0^+$, $g(x)\to+\infty$.
\item $x\to+\infty$ : $g(x)\to0^+$.
\end{itemize}
\underline{Asymptotes.} La droite $x=1$ est asymptote verticale ; l'axe des abscisses $y=0$ est asymptote horizontale en $+\infty$. (En $0$, il n'y a pas d'asymptote : $g$ se prolonge par continuité par $0$.)

\textbf{2) Existence de $J(x)=\displaystyle\int_x^{2x}g(t)\,dt$ pour $x\in D$.}

Pour $x>0$, $J(x)$ est l'intégrale d'une fonction continue sur un segment si et seulement si $[x,2x]\subset]0,1[$ ou $[x,2x]\subset]1,+\infty[$, c'est-à-dire $2x<1$ ou $x>1$ : $x\in D=]0,\frac12[\cup]1,+\infty[$. Dans ce cas $J(x)$ existe et, $G$ étant une primitive de $g$ sur l'intervalle contenant $[x,2x]$, $J(x)=G(2x)-G(x)$.

(Si $x\in[\frac12,1]$, l'intervalle $[x,2x]$ contient $1$, où $g$ n'est pas définie ; on peut montrer que l'intégrale généralisée diverge, car $g(t)\sim\frac1{t-1}$ en $1$.)

\textbf{3) Dérivabilité et variations de $f$ sur $D$.}

Sur chacun des intervalles de $D$, $f(x)=G(2x)-G(x)$ est dérivable (composée et différence de fonctions dérivables) et
\[
f'(x)=2g(2x)-g(x)=\frac2{\ln(2x)}-\frac1{\ln x}=\frac{2\ln x-\ln(2x)}{\ln x\,\ln(2x)}=\frac{\ln x-\ln2}{\ln x\,\ln(2x)}=\boxed{\frac{\ln(x/2)}{\ln x\,\ln(2x)}}.
\]
\underline{Signe.}
\begin{itemize}
\item Pour $0<x<\frac12$ : $\ln x<0$, $\ln(2x)<0$ (car $2x<1$), donc le dénominateur est $>0$ ; et $\ln\frac x2<0$. Donc $f'(x)<0$ : \textbf{$f$ est strictement décroissante sur $]0,\frac12[$}.
\item Pour $x>1$ : $\ln x>0$ et $\ln(2x)>0$ ; $f'(x)$ a le signe de $\ln\frac x2$ : $f'<0$ sur $]1,2[$, $f'(2)=0$, $f'>0$ sur $]2,+\infty[$. \textbf{$f$ décroît sur $]1,2]$ et croît sur $[2,+\infty[$}.
\end{itemize}
On note aussi que $f'(x)\to0$ quand $x\to0^+$ (le numérateur est équivalent à $\ln x$ et le dénominateur à $\ln^2x$).

\textbf{4) Encadrement et comportement en $0$.}

Soit $x\in D$. Sur $[x,2x]$, $g$ est décroissante (question 1), donc pour $t\in[x,2x]$ : $g(2x)\le g(t)\le g(x)$. En intégrant sur $[x,2x]$, de longueur $x$ :
\[
\boxed{\frac{x}{\ln(2x)}\le f(x)\le\frac{x}{\ln x}},\qquad\text{soit}\qquad\frac1{\ln(2x)}\le\frac{f(x)}x\le\frac1{\ln x}.
\]
Quand $x\to0^+$, $\frac1{\ln(2x)}\to0$ et $\frac1{\ln x}\to0$. Par encadrement :
\[
\lim_{x\to0^+}\frac{f(x)}{x}=0\qquad\text{et donc}\qquad\lim_{x\to0^+}f(x)=\lim_{x\to0^+}x\cdot\frac{f(x)}{x}=0 .
\]
\underline{Continuité en 0} : $\lim\limits_{0^+}f=0=f(0)$, $f$ est continue (à droite) en $0$.

\underline{Dérivabilité en 0} : $\dfrac{f(x)-f(0)}{x-0}=\dfrac{f(x)}{x}\to0$ : $f$ est dérivable (à droite) en $0$ et $f'(0)=0$ ; la courbe admet en $0$ une tangente horizontale (en accord avec $\lim_{0^+}f'=0$).

\textbf{5) Comportement en $+\infty$.} D'après l'encadrement : $f(x)\ge\dfrac{x}{\ln(2x)}\to+\infty$ (croissances comparées), donc $\boxed{\lim\limits_{+\infty}f=+\infty}$ ; et $0\le\dfrac{f(x)}x\le\dfrac1{\ln x}\to0$ (pour $x>1$), donc $\boxed{\lim\limits_{+\infty}\frac{f(x)}x=0}$ : branche parabolique de direction $(Ox)$.

\textbf{6) Limite en $\frac12$.}

\underline{Étude de $h(u)=\ln u-2u+2$ sur $]0,1]$.} $h'(u)=\frac1u-2=\frac{1-2u}{u}$ : $h$ est strictement croissante sur $]0,\frac12]$ et strictement décroissante sur $[\frac12,1]$. De plus $\lim_{0^+}h=-\infty$, $h(\frac12)=-\ln2+1>0$ et $h(1)=0$.
\begin{itemize}
\item Sur $]0,\frac12]$ : $h$ est continue et strictement croissante de $-\infty$ à $1-\ln2>0$ ; par le théorème de la bijection, il existe un unique $\alpha\in]0,\frac12[$ tel que $h(\alpha)=0$ (numériquement $\alpha\approx0{,}203$).
\item Sur $[\frac12,1]$ : $h$ décroît de $1-\ln2$ à $0$, donc $h\ge0$ et ne s'annule qu'en $1$.
\end{itemize}
Ainsi $h(u)\ge0$ pour tout $u\in[\alpha,1]$, c'est-à-dire $\ln u\ge2u-2$.

\underline{Majoration.} Soit $x\in[\alpha,\frac12[$ et $t\in[x,2x]\subset[\alpha,1[$. On a $2t-2\le\ln t<0$ ; deux nombres négatifs étant rangés dans l'ordre inverse de leurs inverses : $\dfrac1{\ln t}\le\dfrac1{2t-2}$. En intégrant :
\[
f(x)\le\int_x^{2x}\frac{dt}{2(t-1)}=\frac12\Big[\ln|t-1|\Big]_x^{2x}=\frac12\ln\frac{1-2x}{1-x}=\frac12\ln\Big(\frac{2x-1}{x-1}\Big).
\]
Quand $x\to\frac12^-$, $\frac{1-2x}{1-x}\to0^+$, donc le majorant tend vers $-\infty$, et
\[
\boxed{\lim_{x\to\frac12^-}f(x)=-\infty}.
\]

\textbf{7) Limite en $1^+$.}

La fonction $v(u)=u-1-\ln u$ vérifie $v'(u)=1-\frac1u\ge0$ pour $u\ge1$ et $v(1)=0$ ; donc $v\ge0$, soit $\ln u\le u-1$ pour $u\ge1$.

Soit $x>1$ et $t\in[x,2x]$ : $0<\ln t\le t-1$, donc $\dfrac1{\ln t}\ge\dfrac1{t-1}$. En intégrant :
\[
f(x)\ge\int_x^{2x}\frac{dt}{t-1}=\ln\frac{2x-1}{x-1}\xrightarrow[x\to1^+]{}+\infty,\qquad\boxed{\lim_{x\to1^+}f(x)=+\infty}.
\]

\medskip\textbf{Exercice 3}

\textbf{1) Comparaison de $A$ et $B$.} Sur $[0,\frac\pi4]$, $\cos x>0$ et $\cos(\frac\pi4-x)>0$ : les intégrales existent. Dans $B$, posons $u=\frac\pi4-x$ ($du=-dx$ ; $x=0\mapsto u=\frac\pi4$, $x=\frac\pi4\mapsto u=0$) :
\[
B=\int_{\pi/4}^0\ln(\cos u)\,(-du)=\int_0^{\pi/4}\ln(\cos u)\,du=A .
\]
\[
\boxed{A=B}.
\]

\textbf{2) Calcul de $I=\displaystyle\int_0^{\pi/4}\ln(1+\tan x)\,dx$.}

Pour $x\in[0,\frac\pi4]$ :
\[
1+\tan x=\frac{\cos x+\sin x}{\cos x}\qquad\text{et}\qquad\cos x+\sin x=\sqrt2\cos\Big(\frac\pi4-x\Big)
\]
(car $\cos(\frac\pi4-x)=\frac{\sqrt2}2\cos x+\frac{\sqrt2}2\sin x$). Donc
\[
\ln(1+\tan x)=\ln\sqrt2+\ln\cos\Big(\frac\pi4-x\Big)-\ln(\cos x).
\]
En intégrant sur $[0,\frac\pi4]$ :
\[
I=\frac\pi4\ln\sqrt2+B-A=\frac\pi4\cdot\frac{\ln2}{2}+0,\qquad\boxed{I=\frac{\pi\ln2}{8}\approx\frac{3{,}14\times0{,}69}{8}\approx0{,}27}.
\]
